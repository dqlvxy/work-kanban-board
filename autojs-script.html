/**
 * 手机自动投递 · 第 14 步：先认加号，再点同位纸飞机
 * 空框时右侧是圆圈加号，有字同位置变纸飞机。先清空，扫出加号坐标，填字，再点同一坐标。
 */

var 话术 = "孙先生您好，我是杜全龙，18 年 ICT/FCT 测试治具与 FPC 连接器治具机械设计经验，熟练使用 SolidWorks，做过手机 FCT 测试设备、保压盒缓存机构、双车床机器人上下料等项目，服务过华为、OPPO、比亚迪、小米等客户。看到贵司招机械/结构工程师，5-10 年经验、大专，跟我的履历对得上。方便的话我把简历发您看下？";

auto.waitFor();
sleep(800);

var 报告 = "";
launchApp("BOSS直聘");
sleep(4500);

if (currentPackage() != "com.hpbr.bosszhipin") {
    setClip("不在 BOSS 前台");
    toastLog("不在 BOSS");
    exit();
}

var 输入 = className("android.widget.EditText").findOne(6000);
if (!输入) {
    setClip("没找到输入框");
    toastLog("没找到输入框");
    exit();
}

function 发出去了() {
    return !!textContains("杜全龙").className("android.widget.TextView").findOnce(2500);
}

// 扫右侧那个圆形按钮（加号/纸飞机），取最靠右的小节点
function 找右侧按钮(框) {
    var 全部 = classNameMatches(/.+/).find();
    var 最佳 = null;
    for (var i = 0; i < 全部.size(); i++) {
        var w = 全部.get(i);
        var b = w.bounds();
        if (b.width() < 30 || b.width() > 280) continue;
        if (b.height() < 30 || b.height() > 280) continue;
        if (b.centerX() < device.width * 0.80) continue;
        if (Math.abs(b.centerY() - 框.centerY()) > 170) continue;
        if (!最佳 || b.centerX() > 最佳.centerX()) 最佳 = b;
    }
    return 最佳;
}

// 1) 清空
输入.click();
sleep(1000);
try { 输入.setText(""); } catch (e) {}
sleep(600);
try { 输入.setText(""); } catch (e) {}
sleep(1600);
报告 += "已清空\n";

var 框 = 输入.bounds();
报告 += "输入框: " + 框.left + "," + 框.top + " " + 框.width() + "x" + 框.height() + "\n";

// 2) 认加号
var 钮 = 找右侧按钮(框);
var 目标X, 目标Y;
if (钮) {
    目标X = 钮.centerX();
    目标Y = 钮.centerY();
    报告 += "扫到圆钮(加号): " + 目标X + "," + 目标Y + "  " + 钮.width() + "x" + 钮.height() + "\n";
} else {
    目标X = device.width - 100;
    目标Y = 框.centerY();
    报告 += "没扫到节点，用几何位置: " + 目标X + "," + 目标Y + "\n";
}

// 3) 填话术
try { 输入.setText(话术); } catch (e) { 报告 += "setText 异常: " + e + "\n"; }
sleep(2200);
报告 += "已填话术，准备点同位置\n";

var 已发 = false;
for (var r = 1; r <= 3 && !已发; r++) {
    click(目标X, 目标Y);
    sleep(2000);
    已发 = 发出去了();
    报告 += "第" + r + "次 click(" + 目标X + "," + 目标Y + "): " + 已发 + "\n";
    if (已发) break;
    try { swipe(目标X - 4, 目标Y - 4, 目标X + 4, 目标Y + 4, 80); } catch (e) {}
    sleep(2000);
    已发 = 发出去了();
    报告 += "第" + r + "次 swipe: " + 已发 + "\n";
}

if (!已发) {
    var 笑X = device.width - 282, 笑Y = 框.centerY();
    var 前 = classNameMatches(/.+/).find().size();
    click(笑X, 笑Y);
    sleep(1800);
    var 后 = classNameMatches(/.+/).find().size();
    报告 += "自检 点笑脸 节点 " + 前 + "→" + 后 + "\n";
    报告 += (后 > 前 + 10) ? "手势能传进 App\n" : "手势被忽略\n";
    输入.click(); sleep(600);
}

sleep(800);
setClip(报告);
toastLog(已发 ? "发出去了" : "没发出去");
