脚本全文（按钮失灵时手动复制）
/**
 * 手机自动投递 · 第 12 步：确认有字 → 点纸飞机 → 手势自检
 * 空框时右侧是圆圈加号，有字才变纸飞机，位置固定：
 * x = 屏宽 - 100，y = 输入框垂直中心。
 */

var 话术 = "孙先生您好，我是杜全龙，18 年 ICT/FCT 测试治具与 FPC 连接器治具机械设计经验，熟练使用 SolidWorks，做过手机 FCT 测试设备、保压盒缓存机构、双车床机器人上下料等项目，服务过华为、OPPO、比亚迪、小米等客户。看到贵司招机械/结构工程师，5-10 年经验、大专，跟我的履历对得上。方便的话我把简历发您看下？";

auto.waitFor();
sleep(800);

var 报告 = "";
var 拉起 = launchApp("BOSS直聘");
sleep(4500);

if (currentPackage() != "com.hpbr.bosszhipin") {
    setClip("不在 BOSS 前台");
    toastLog("不在 BOSS");
    exit();
}

var 输入 = className("android.widget.EditText").findOne(6000);
if (!输入) {
    setClip("没找到输入框");
    toastLog("没找到输入框");
    exit();
}

function 字数() {
    return ((输入.text() || "").trim()).length;
}

// 1) 把话术弄进去，多路径，直到字数过关
输入.click();
sleep(1000);
报告 += "初始字数: " + 字数() + "\n";

if (字数() < 100) {
    try { 输入.setText(话术); } catch (e) {}
    sleep(1600);
    报告 += "setText 后: " + 字数() + "\n";

    if (字数() < 100) {
        setClip(话术);
        sleep(400);
        try {
            var 框0 = 输入.bounds();
            longClick(框0.centerX(), 框0.centerY());
            sleep(1600);
            var 粘 = textContains("粘贴").findOne(2000);
            if (粘) { 粘.click(); sleep(1400); }
        } catch (e) { 报告 += "粘贴异常: " + e + "\n"; }
        报告 += "粘贴后: " + 字数() + "\n";
    }

    if (字数() < 100) {
        输入.click();
        sleep(600);
        try { 输入.setText(话术); } catch (e) {}
        sleep(1500);
        报告 += "二次 setText 后: " + 字数() + "\n";
    }
}

if (字数() < 100) {
    setClip(报告 + "话术进不去，停止");
    toastLog("话术进不去");
    exit();
}

sleep(1200);
var 框 = 输入.bounds();
var 目标X = device.width - 100;
var 目标Y = 框.centerY();
报告 += "输入框: " + 框.left + "," + 框.top + " " + 框.width() + "x" + 框.height() + "\n";
报告 += "目标点: " + 目标X + "," + 目标Y + "\n";

var 已发 = false;

// 2) 点纸飞机：click 与 swipe 交替，各点两轮
for (var r = 0; r < 2 && !已发; r++) {
    click(目标X, 目标Y);
    sleep(1800);
    报告 += "第" + (r + 1) + "次 click 后字数: " + 字数() + "\n";
    if (字数() < 8) { 已发 = true; 报告 += "click 发出去\n"; break; }

    try { swipe(目标X - 4, 目标Y - 4, 目标X + 4, 目标Y + 4, 80); } catch (e) {}
    sleep(1800);
    报告 += "第" + (r + 1) + "次 swipe 后字数: " + 字数() + "\n";
    if (字数() < 8) { 已发 = true; 报告 += "swipe 发出去\n"; break; }
}

// 3) 手势自检：点笑脸，看手势到底能不能传进去
if (!已发) {
    var 笑X = device.width - 282;
    var 笑Y = 框.centerY();
    var 前字 = 字数();
    var 前节点 = classNameMatches(/.+/).find().size();
    click(笑X, 笑Y);
    sleep(1600);
    var 后字 = 字数();
    var 后节点 = classNameMatches(/.+/).find().size();
    报告 += "自检 点笑脸: 字数 " + 前字 + "→" + 后字 + "  节点 " + 前节点 + "→" + 后节点 + "\n";
    if (后节点 > 前节点 + 10 || 后字 != 前字) {
        报告 += "结论: 手势可以传进 App\n";
        输入.click();
        sleep(800);
        try { back(); } catch (e) {}
    } else {
        报告 += "结论: 手势被 App 忽略，坐标点无效\n";
    }
}

sleep(800);
setClip(报告);
toastLog(已发 ? "发出去了" : "没发出去，看报告");
点失败它会自动测一次笑脸，判断手势能不能传进 App
