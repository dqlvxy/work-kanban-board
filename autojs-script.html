/**
 * 手机自动投递 · 第 28 步：先认页面，再决定点哪一步
 * 首页 → 点底部「消息」；消息列表 → 点「孙先生」；已在聊天页 → 前两步都不走
 */

var PKG = "com.hpbr.bosszhipin";
var 话术 = "孙先生您好，我是杜全龙，18 年 ICT/FCT 测试治具与 FPC 连接器治具机械设计经验，熟练使用 SolidWorks，做过手机 FCT 测试设备、保压盒缓存机构、双车床机器人上下料等项目，服务过华为、OPPO、比亚迪、小米等客户。看到贵司招机械/结构工程师，5-10 年经验、大专，跟我的履历对得上。方便的话我把简历发您看下？";

auto.waitFor();
sleep(700);
var 报告 = "";

function 宽(b) { return b.right - b.left; }
function 高(b) { return b.bottom - b.top; }
function 心X(b) { return (b.left + b.right) / 2; }
function 心Y(b) { return (b.top + b.bottom) / 2; }
function 屏H() { return device.height; }

// ———— 三种页面判据 ————

// 聊天页：底部有输入框（top 在屏幕 70% 以下位置）
function 在聊天页() {
    var e = className("android.widget.EditText").findOnce();
    return e != null && e.bounds().top > 屏H() * 0.70;
}

// 消息列表：屏上出现「孙先生」，且没有底部输入框
function 见孙先生() {
    var t = textContains("孙先生").findOnce(1500);
    if (!t) return null;
    if (在聊天页()) return null;   // 聊天页顶部也有孙先生，排除掉
    return t;
}

// 底部页签：文字在屏幕 85% 以下
function 找页签(名) {
    var hit = null;
    text(名).find().forEach(function (n) {
        if (心Y(n.bounds()) > 屏H() * 0.85) hit = n;
    });
    return hit;
}

// 文字节点自己不可点，往上找可点的祖先
function 点祖先(n, 层) {
    var cur = n;
    for (var k = 0; k < 层; k++) {
        if (!cur) return false;
        if (cur.clickable()) { cur.click(); return true; }
        cur = cur.parent();
    }
    return false;
}

function 点文字(n) {
    if (!n) return false;
    if (点祖先(n, 10)) return true;
    return n.click();
}

launchApp("BOSS直聘");
sleep(4500);
报告 += "启动后前台: " + currentPackage() + "\n";

// ———— 状态机：每轮认一次页面，只做一步 ————
var 用了 = "";
for (var step = 1; step <= 6; step++) {
    if (currentPackage() != PKG) {
        报告 += step + ": 不在 BOSS，重新拉起\n";
        launchApp("BOSS直聘");
        sleep(4000);
        continue;
    }

    if (在聊天页()) {
        报告 += step + ": 已在孙先生聊天页，前两步跳过\n";
        用了 = "直接进聊天页";
        break;
    }

    var 孙 = 见孙先生();
    if (孙) {
        报告 += step + ": 消息列表，点孙先生那一栏\n";
        点文字(孙);
        sleep(2600);
        用了 = "消息列表→点孙先生";
        continue;
    }

    var 消息 = 找页签("消息");
    if (消息) {
        报告 += step + ": 首页，点底部消息\n";
        点文字(消息);
        sleep(2400);
        用了 = "首页→点消息";
        continue;
    }

    报告 += step + ": 认不出界面，back 一层\n";
    back();
    sleep(1600);
    if (currentPackage() != PKG) { launchApp("BOSS直聘"); sleep(4000); }
}

if (!在聊天页()) {
    setClip(报告 + "\n最终没进到孙先生聊天页，脚本停手。");
    toastLog("没进聊天页");
    exit();
}

var 输入 = className("android.widget.EditText").findOne(4000);
var ib = 输入.bounds();
报告 += "输入框: " + ib.left + "," + ib.top + " " + 宽(ib) + "x" + 高(ib) + "\n";

function 面板开了() { return textContains("语音通话").findOnce(900) != null; }
function 发出去了() { return !!textContains("杜全龙").className("android.widget.TextView").findOnce(2000); }

// 右侧候选：屏幕上右半区偏下的所有可点小方块
function 右侧候选() {
    var arr = [];
    packageName(PKG).clickable(true).find().forEach(function (n) {
        var b = n.bounds();
        var w = 宽(b), h = 高(b);
        if (w <= 0 || h <= 0 || w > 320 || h > 320) return;
        if (心X(b) > device.width * 0.60 && 心Y(b) > 屏H() * 0.42) arr.push(n);
    });
    return arr;
}

function 列清单(标签) {
    var arr = 右侧候选(), out = [];
    arr.forEach(function (n) {
        var b = n.bounds();
        out.push("[" + (n.text() || n.desc() || "无字") + "@" + 心X(b) + "," + 心Y(b) + " " + 宽(b) + "x" + 高(b) + "]");
    });
    报告 += "\n" + 标签 + "：" + arr.length + " 个\n" + out.join("  ") + "\n";
    return arr;
}

// 空框态先找加号真身：哪个点下去能弹出「语音通话」面板
var 空框 = 列清单("空框状态");
报告 += "空框时面板: " + 面板开了() + "\n";

var 命中 = null;
for (var i = 0; i < 空框.length; i++) {
    var n = 空框[i], b = n.bounds();
    if (心Y(b) < 屏H() * 0.6) continue;       // 上部的不试
    if (心Y(b) < ib.top - 160) continue;      // 离输入框太远的不试
    报告 += "试 @" + 心X(b) + "," + 心Y(b) + " → ";
    n.click();
    sleep(1200);
    if (面板开了()) { 报告 += "命中，这就是加号\n"; 命中 = n; break; }
    报告 += "无反应\n";
}

if (命中) {
    var hb = 命中.bounds();
    报告 += "加号: " + 心X(hb) + "," + 心Y(hb) + " " + 宽(hb) + "x" + 高(hb) + "\n";
    var arr2 = 右侧候选(), mr = null;
    arr2.forEach(function (n) { if (!mr || 心X(n.bounds()) > 心X(mr.bounds())) mr = n; });
    if (mr) { mr.click(); sleep(1200); }
    报告 += "收面板后: " + 面板开了() + "\n";
}

// 填话术
输入.click();
sleep(900);
try { 输入.setText(话术); } catch (e) { 报告 += "setText 异常: " + e + "\n"; }
sleep(2400);
var nb = 输入.bounds();
报告 += "\n填字后输入框: " + nb.left + "," + nb.top + " " + 宽(nb) + "x" + 高(nb) + "（位移 " + (nb.top - ib.top) + "）\n";
列清单("填字后状态");

var 已发 = false;
if (命中) {
    var 原Y = 心Y(命中.bounds());
    for (var r = 1; r <= 3 && !已发; r++) {
        var arr3 = 右侧候选(), 目标 = null, 最小差 = 99999;
        arr3.forEach(function (n) {
            var b = n.bounds();
            var d = Math.abs(心Y(b) - 原Y);
            if (d < 最小差) { 最小差 = d; 目标 = n; }
        });
        if (!目标) { 报告 += "第" + r + "次无候选\n"; break; }
        var tb = 目标.bounds();
        报告 += "第" + r + "次点 @" + 心X(tb) + "," + 心Y(tb) + "（距加号位 " + 最小差 + "）→ ";
        目标.click();
        sleep(2100);
        已发 = 发出去了();
        报告 += "发出 " + 已发 + "\n";
        if (!已发 && 面板开了()) 报告 += "  （又把面板点开了）\n";
    }
}

sleep(700);
setClip(报告);
toastLog(已发 ? "发出去了" : "没发出去");
