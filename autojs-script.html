/**
 * 手机自动投递 · 诊断版 v2
 * 只读不点，用来判断无障碍服务到底通没通。
 */
auto.waitFor();
sleep(1500);

toastLog("无障碍服务: " + (auto.service ? "已开启" : "未开启"));
sleep(2500);

toastLog("前台包名: " + currentPackage());
sleep(2500);

var 全部 = classNameMatches(/.+/).find();
toastLog("界面节点总数: " + 全部.size());
sleep(2000);

var n = 0;
for (var i = 0; i < 全部.size(); i++) {
    var w = 全部.get(i);
    var 文 = (w.text() || "").trim();
    var 述 = (w.desc() || "").trim();
    if (!文 && !述) continue;
    var b = w.bounds();
    console.log(w.className().replace("android.widget.", "").replace("android.view.", "") +
        " | " + 文 + " | " + 述 + " | " +
        b.left + "," + b.top + " " + b.width() + "x" + b.height());
    n++;
}
toastLog("扫描结束，含文字节点 " + n + " 个，把日志复制发我");
复制失败多半是浏览器拦了剪贴板，点「全选」再长按复制一样能用
