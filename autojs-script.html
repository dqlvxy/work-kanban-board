/**
 * 手机自动投递 · 第 32 步：逐行点开，看顶部名字是不是孙先生
 * v31 已读到消息列表：会话行是 android.view.ViewGroup 1152x228，@576,573 / 801 / 1029 / 1257 …
 * 除第一行有文字，其余行文字读不到，只能逐行点开用聊天页顶部名字判断。
 */

var PKG = "com.hpbr.bosszhipin";
var 话术 = "孙先生您好，我是杜全龙，18 年 ICT/FCT 测试治具与 FPC 连接器治具机械设计经验，熟练使用 SolidWorks，做过手机 FCT 测试设备、保压盒缓存机构、双车床机器人上下料等项目，服务过华为、OPPO、比亚迪、小米等客户。看到贵司招机械/结构工程师，5-10 年经验、大专，跟我的履历对得上。方便的话我把简历发您看下？";

auto.waitFor();
sleep(600);
var 报告 = "";

function 宽(b) { return b.right - b.left; }
function 高(b) { return b.bottom - b.top; }
function 心X(b) { return (b.left + b.right) / 2; }
function 心Y(b) { return (b.top + b.bottom) / 2; }
function 屏H() { return device.height; }

launchApp("BOSS直聘");
sleep(4500);
if (currentPackage() != PKG) { launchApp("BOSS直聘"); sleep(4000); }
报告 += "前台 " + currentPackage() + "\n";
if (currentPackage() != PKG) { setClip(报告 + "BOSS 没到前台，停手"); toastLog("停手"); exit(); }
sleep(1000);

function 在聊天页() {
    var e = className("android.widget.EditText").findOnce();
    return e != null && e.bounds().top > 屏H() * 0.70;
}

// 聊天页顶部名字
function 顶部称呼() {
    var hit = null;
    textContains("先生").find().forEach(function (n) {
        if (心Y(n.bounds()) < 屏H() * 0.20) hit = n;
    });
    if (!hit) {
        textContains("女士").find().forEach(function (n) {
            if (心Y(n.bounds()) < 屏H() * 0.20) hit = n;
        });
    }
    return hit ? hit.text() : "";
}

// 会话行：宽度撑满、行高 140-300 的 ViewGroup
function 会话行组() {
    var arr = [];
    className("android.view.ViewGroup").find().forEach(function (n) {
        var b = n.bounds();
        var w = 宽(b), h = 高(b);
        if (w > 1000 && h > 140 && h < 320 && 心Y(b) > 屏H() * 0.18) arr.push(n);
    });
    arr.sort(function (a, b) { return 心Y(a.bounds()) - 心Y(b.bounds()); });
    return arr;
}

function 回列表() {
    for (var k = 0; k < 2; k++) {
        back();
        sleep(1500);
        if (currentPackage() != PKG) { launchApp("BOSS直聘"); sleep(4000); }
        if (!在聊天页()) return true;
    }
    return !在聊天页();
}

// ———— 找孙先生 ————
var 找到 = 在聊天页() && 顶部称呼().indexOf("孙") >= 0;
if (在聊天页() && !找到) { 报告 += "当前在别人聊天页，退回列表\n"; 回列表(); }

if (!找到) {
    var 行 = 会话行组();
    报告 += "会话行 " + 行.length + " 个\n";
    for (var i = 0; i < 行.length && i < 10; i++) {
        var b = 行[i].bounds();
        报告 += "点第" + (i + 1) + "行 @" + Math.round(心Y(b)) + " → ";
        行[i].click();
        sleep(2700);
        if (!在聊天页()) { 报告 += "没进聊天页\n"; continue; }
        var 名 = 顶部称呼();
        报告 += "进了，顶部=" + 名 + "\n";
        if (名.indexOf("孙") >= 0) { 找到 = true; 报告 += "  这就是孙先生\n"; break; }
        报告 += "  不是，退回列表\n";
        if (!回列表()) { 报告 += "  退不回列表，停手\n"; break; }
    }
}

if (!找到) {
    setClip(报告 + "\n十行里没找到孙先生");
    toastLog("没找到孙先生");
    exit();
}

// ———— 进到聊天页，开始干活 ————
var 输入 = className("android.widget.EditText").findOne(4000);
if (!输入) { setClip(报告 + "\n聊天页里没找到输入框"); toastLog("没输入框"); exit(); }
var ib = 输入.bounds();
报告 += "\n输入框: " + ib.left + "," + ib.top + " " + 宽(ib) + "x" + 高(ib) + "\n";

function 面板开了() { return textContains("语音通话").findOnce(900) != null; }
function 发出去了() { return !!textContains("杜全龙").className("android.widget.TextView").findOnce(2000); }

function 右侧候选() {
    var arr = [];
    classNameMatches(/.*/).find().forEach(function (n) {
        try {
            var b = n.bounds();
            var w = 宽(b), h = 高(b);
            if (w <= 0 || h <= 0 || w > 320 || h > 320) return;
            if (心X(b) > device.width * 0.60 && 心Y(b) > 屏H() * 0.42) arr.push(n);
        } catch (e) { }
    });
    return arr;
}

function 列清单(标签) {
    var arr = 右侧候选(), out = [];
    arr.forEach(function (n) {
        var b = n.bounds();
        out.push("[" + (n.text() || n.desc() || "无字") + "@" + Math.round(心X(b)) + "," + Math.round(心Y(b)) + " " + 宽(b) + "x" + 高(b) + "]");
    });
    报告 += "\n" + 标签 + "：" + arr.length + " 个\n" + out.join("  ") + "\n";
    return arr;
}

var 空框 = 列清单("空框状态");
报告 += "空框时面板: " + 面板开了() + "\n";

var 命中 = null;
for (var j = 0; j < 空框.length; j++) {
    var n2 = 空框[j], b2 = n2.bounds();
    if (心Y(b2) < 屏H() * 0.60) continue;
    if (心Y(b2) < ib.top - 160) continue;
    报告 += "试 @" + Math.round(心X(b2)) + "," + Math.round(心Y(b2)) + " → ";
    n2.click();
    sleep(1200);
    if (面板开了()) { 报告 += "命中，这就是加号\n"; 命中 = n2; break; }
    报告 += "无反应\n";
}

if (命中) {
    var hb = 命中.bounds();
    报告 += "加号: " + Math.round(心X(hb)) + "," + Math.round(心Y(hb)) + "\n";
    var a9 = 右侧候选(), mr = null;
    a9.forEach(function (n) { if (!mr || 心X(n.bounds()) > 心X(mr.bounds())) mr = n; });
    if (mr) { mr.click(); sleep(1200); }
    报告 += "收面板后: " + 面板开了() + "\n";
}

输入.click();
sleep(900);
try { 输入.setText(话术); } catch (e) { 报告 += "setText 异常: " + e + "\n"; }
sleep(2400);
var nb = 输入.bounds();
报告 += "\n填字后输入框: " + nb.left + "," + nb.top + "（位移 " + (nb.top - ib.top) + "）\n";
列清单("填字后状态");

var 已发 = false;
if (命中) {
    var 原Y = 心Y(命中.bounds());
    for (var r = 1; r <= 3 && !已发; r++) {
        var a3 = 右侧候选(), 目标 = null, 最小差 = 99999;
        a3.forEach(function (n) {
            var b = n.bounds();
            var d = Math.abs(心Y(b) - 原Y);
            if (d < 最小差) { 最小差 = d; 目标 = n; }
        });
        if (!目标) { 报告 += "第" + r + "次无候选\n"; break; }
        var tb = 目标.bounds();
        报告 += "第" + r + "次点 @" + Math.round(心X(tb)) + "," + Math.round(心Y(tb)) + " → ";
        目标.click();
        sleep(2100);
        已发 = 发出去了();
        报告 += "发出 " + 已发 + "\n";
    }
}

sleep(600);
setClip(报告);
toastLog(已发 ? "发出去了" : "没发出去");
