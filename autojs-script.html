脚本全文（按钮失灵时手动复制）
/**
 * 手机自动投递 · 第 6 步：真点一次「立即沟通」
 * 只点这一下，不自动发消息。点完把新界面结构复制到剪贴板。
 * 用法：先在 BOSS 停在某个岗位详情页，按 Home 退出，再运行本脚本。
 */

function 扫描() {
    var 全部 = classNameMatches(/.+/).find();
    var 行 = [];
    for (var i = 0; i < 全部.size(); i++) {
        var w = 全部.get(i);
        var 文 = (w.text() || "").trim();
        var 述 = (w.desc() || "").trim();
        if (!文 && !述) continue;
        var b = w.bounds();
        行.push(w.className().replace("android.widget.", "").replace("android.view.", "") +
            " | " + 文 + " | " + 述 + " | " +
            b.left + "," + b.top + " " + b.width() + "x" + b.height());
    }
    return "节点总数: " + 全部.size() + "  含文字: " + 行.length + "\n----\n" + 行.join("\n");
}
