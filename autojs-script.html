脚本全文（按钮失灵时手动复制）
/**
 * 手机自动投递 · 第 3 步：扫 BOSS 真实界面
 * 只读不点。脚本自动打开 BOSS 直聘并扫描，结果存到 Download/boss_scan.txt
 */
auto.waitFor();
sleep(1000);

var 目标 = "";
if (launchApp("BOSS直聘")) {
    目标 = "BOSS直聘";
} else if (launchApp("猎聘")) {
    目标 = "猎聘";
}
toastLog("打开结果: " + (目标 || "两个都没打开"));

if (目标) {
    sleep(8000);
    var 包 = currentPackage();
    toastLog("当前前台: " + 包);
    sleep(1500);

    var 全部 = classNameMatches(/.+/).find();
    var 行 = [];
    for (var i = 0; i < 全部.size(); i++) {
        var w = 全部.get(i);
        var 文 = (w.text() || "").trim();
        var 述 = (w.desc() || "").trim();
        if (!文 && !述) continue;
        var b = w.bounds();
        行.push(w.className().replace("android.widget.", "").replace("android.view.", "") +
            " | " + 文 + " | " + 述 + " | " +
            b.left + "," + b.top + " " + b.width() + "x" + b.height());
    }

    var 内容 = "包名: " + 包 + "\n节点总数: " + 全部.size() +
        "\n含文字: " + 行.length + "\n----\n" + 行.join("\n");
    var 路径 = "/sdcard/Download/boss_scan.txt";
    files.write(路径, 内容);
    toastLog("已写入 boss_scan.txt，共 " + 行.length + " 条");
}
跑完手机可能停在 BOSS 界面，按返回退出再回 AutoJs6 取文件
