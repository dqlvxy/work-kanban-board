脚本全文（按钮失灵时手动复制）
/**
 * 手机自动投递 · 第 18 步：竖线盲扫找加号，认出面板记坐标，再填话术点同一点
 */

var 话术 = "孙先生您好，我是杜全龙，18 年 ICT/FCT 测试治具与 FPC 连接器治具机械设计经验，熟练使用 SolidWorks，做过手机 FCT 测试设备、保压盒缓存机构、双车床机器人上下料等项目，服务过华为、OPPO、比亚迪、小米等客户。看到贵司招机械/结构工程师，5-10 年经验、大专，跟我的履历对得上。方便的话我把简历发您看下？";

auto.waitFor();
sleep(800);

var 报告 = "";
launchApp("BOSS直聘");
sleep(4500);
报告 += "包名: " + currentPackage() + "\n";

var 输入 = className("android.widget.EditText").findOne(6000);
if (!输入) {
    setClip("没找到输入框，先手动进到孙先生聊天页再跑");
    toastLog("没找到输入框");
    exit();
}

function 面板出现了() {
    return textContains("语音通话").findOnce(1200) != null ||
           textContains("语音").findOnce(1200) != null;
}

function 发出去了() {
    return !!textContains("杜全龙").className("android.widget.TextView").findOnce(2200);
}

输入.click();
sleep(1200);

var 框 = 输入.bounds();
报告 += "输入框: " + 框.left + "," + 框.top + " " + 框.width() + "x" + 框.height() + "\n";

var X = device.width - 100;
报告 += "竖线 x=" + X + "\n";

var 找到Y = -1;
var 试过 = 0;

for (var y = 框.top - 60; y <= 框.top + 框.height() + 60; y += 30) {
    click(X, y);
    sleep(1250);
    试过++;
    if (面板出现了()) {
        找到Y = y;
        报告 += "命中加号 @ " + X + "," + y + "  试了 " + 试过 + " 次\n";
        break;
    }
    // 若误弹表情面板，点输入框收回
    if (textContains("表情").exists()) {
        输入.click();
        sleep(900);
    }
}

if (找到Y < 0) {
    setClip(报告 + "竖线扫完 " + 试过 + " 个点，加号毫无反应 → 坐标手势确实被系统拦了");
    toastLog("加号扫不到响应");
    exit();
}

// 关掉面板（同一点此时是 ×）
click(X, 找到Y);
sleep(1400);
if (面板出现了()) {
    back();
    sleep(1200);
    报告 += "面板没关掉，已按返回\n";
} else {
    报告 += "面板已关闭\n";
}

// 填话术
输入.click();
sleep(900);
try { 输入.setText(话术); } catch (e) { 报告 += "setText 异常: " + e + "\n"; }
sleep(2200);
报告 += "话术已填入\n";

var 已发 = false;
for (var r = 1; r <= 3 && !已发; r++) {
    click(X, 找到Y);
    sleep(2200);
    已发 = 发出去了();
    报告 += "第" + r + "次点发送位: " + 已发 + "\n";
    if (!已发) {
        // 可能把面板又弹开了，先关掉再继续
        if (面板出现了()) {
            click(X, 找到Y);
            sleep(1300);
            报告 += "  守门：面板又开了，已点 × 收起\n";
        }
    }
}

sleep(1000);
setClip(报告);
toastLog(已发 ? "发出去了" : "找到了加号位但没发出去");
扫不到就说明手势真被拦，扫到就能发
