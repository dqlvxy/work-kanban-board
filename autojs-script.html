/**
 * 手机自动投递 · 第 22 步：认最右页签（我的），摸清设置入口
 * 上版教训：底部 4 个页签的 clickable 节点文字是空的，text("我的") 匹配不到；
 * 另外循环时序错了，最后一次 back 完就到列表页，却没再检查一次。
 */

var PKG = "com.hpbr.bosszhipin";
auto.waitFor();
sleep(800);
var 报告 = "";

function 在BOSS() { return currentPackage() === PKG; }

function 回BOSS() {
    for (var i = 0; i < 3 && !在BOSS(); i++) {
        app.launchPackage(PKG);
        sleep(2600);
    }
    return 在BOSS();
}

// 底部最右页签（我的）节点：clickable、在最下、横扫最右
function 找最右页签() {
    var best = null;
    packageName(PKG).clickable(true).find().forEach(function (n) {
        var b = n.bounds();
        if (b.top > device.height * 0.80 && b.width() > 60 && b.height() > 60) {
            if (!best || b.left > best.bounds().left) best = n;
        }
    });
    return best;
}

function 抓页(标签) {
    if (!在BOSS()) { 报告 += "\n===== " + 标签 + " =====\n（不在 BOSS）\n"; return; }
    var 文 = [], 钮 = [];
    packageName(PKG).className("android.widget.TextView").find().forEach(function (n) {
        var t = n.text();
        if (t && t.length) 文.push(t);
    });
    packageName(PKG).find().forEach(function (n) {
        if (!n.clickable()) return;
        var b = n.bounds();
        if (b.width() > 0 && b.height() > 0)
            钮.push((n.desc() || "") + "|" + (n.text() || "") + "@" + (b.left + b.width() / 2) + "," + (b.top + b.height() / 2));
    });
    报告 += "\n===== " + 标签 + " =====\n";
    报告 += "文字 " + 文.length + ": " + 文.join(" / ") + "\n";
    报告 += "可点 " + 钮.length + ": " + 钮.slice(0, 45).join("  ") + "\n";
}

回BOSS();
报告 += "启动后前台: " + currentPackage() + "\n";
sleep(1500);

// 先 back 再检查，最多 5 轮
var 页签 = null;
for (var i = 0; i < 5; i++) {
    页签 = 找最右页签();
    if (页签) { 报告 += "第" + (i + 1) + "次检查见到底部页签\n"; break; }
    报告 += "第" + (i + 1) + "次没见，按返回\n";
    back();
    sleep(1600);
    回BOSS();
}

if (!页签) {
    抓页("当前位置");
    setClip(报告 + "\n五轮都没摸到底部页签");
    toastLog("没摸到底部页签");
    exit();
}

var b = 页签.bounds();
报告 += "最右页签 @ " + (b.left + b.width() / 2) + "," + (b.top + b.height() / 2) + " " + b.width() + "x" + b.height() + "\n";
页签.click();
sleep(2600);
报告 += "点完前台: " + currentPackage() + "\n";
抓页("我的页");

// 我的页里找设置入口
var 候选 = ["设置", "打招呼", "隐私"];
for (var c = 0; c < 候选.length; c++) {
    var n = textContains(候选[c]).findOnce(500);
    if (n) {
        var q = n;
        for (var j = 0; j < 5 && q && !q.clickable(); j++) q = q.parent();
        if (q) {
            q.click();
            sleep(2400);
            报告 += "点了「" + 候选[c] + "」\n";
            抓页("点" + 候选[c] + "之后");
            back();
            sleep(1600);
            回BOSS();
            抓页("回来的页面");
            break;
        }
    }
}

setClip(报告);
toastLog("结构已复制到剪贴板");
