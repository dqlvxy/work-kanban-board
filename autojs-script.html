/**
 * 手机自动投递 · 第 29 步：消息列表认出来了，改用多种策略点「孙先生」那一栏
 * v28 教训：页面已经在消息列表（顶部「聊天」标题、底部有页签），但 textContains("孙先生") 读成空，
 * 判据把页面当首页，于是一直点底部消息、原地空转。
 */

var PKG = "com.hpbr.bosszhipin";
var 话术 = "孙先生您好，我是杜全龙，18 年 ICT/FCT 测试治具与 FPC 连接器治具机械设计经验，熟练使用 SolidWorks，做过手机 FCT 测试设备、保压盒缓存机构、双车床机器人上下料等项目，服务过华为、OPPO、比亚迪、小米等客户。看到贵司招机械/结构工程师，5-10 年经验、大专，跟我的履历对得上。方便的话我把简历发您看下？";

auto.waitFor();
sleep(700);
var 报告 = "";

function 宽(b) { return b.right - b.left; }
function 高(b) { return b.bottom - b.top; }
function 心X(b) { return (b.left + b.right) / 2; }
function 心Y(b) { return (b.top + b.bottom) / 2; }
function 屏H() { return device.height; }

function 在聊天页() {
    var e = className("android.widget.EditText").findOnce();
    return e != null && e.bounds().top > 屏H() * 0.70;
}

// 消息列表：顶部有「聊天」标题（y 在 15% 屏高以内）
function 在消息列表() {
    var t = text("聊天").findOnce(500);
    return t != null && 心Y(t.bounds()) < 屏H() * 0.16;
}

function 找页签(名) {
    var hit = null;
    text(名).find().forEach(function (n) {
        if (心Y(n.bounds()) > 屏H() * 0.85) hit = n;
    });
    return hit;
}

function 点祖先(n, 层) {
    var cur = n;
    for (var k = 0; k < 层; k++) {
        if (!cur) return false;
        if (cur.clickable()) { cur.click(); return true; }
        cur = cur.parent();
    }
    return false;
}
function 点文字(n) {
    if (!n) return false;
    if (n.clickable()) { n.click(); return true; }
    if (点祖先(n, 10)) return true;
    return n.click();
}

// 会话行候选：屏幕上中部、横向撑满的可点节点
function 会话行() {
    var arr = [];
    packageName(PKG).clickable(true).find().forEach(function (n) {
        var b = n.bounds();
        var w = 宽(b), h = 高(b);
        if (w < 500 || h < 60 || h > 400) return;
        if (心Y(b) < 屏H() * 0.16 || 心Y(b) > 屏H() * 0.86) return;
        arr.push(n);
    });
    arr.sort(function (a, b) { return 心Y(a.bounds()) - 心Y(b.bounds()); });
    return arr;
}

function 点孙先生() {
    // 策略 1：文字直接命中
    var words = ["孙先生", "博以", "西安博以"];
    for (var i = 0; i < words.length; i++) {
        var n = textContains(words[i]).findOnce(800) || descContains(words[i]).findOnce(600);
        if (!n) n = textContains(words[i]).findOnce(300);
        if (n) {
            if (点文字(n)) return "文字命中：" + words[i];
        }
    }
    // 策略 2：按行位置点（列表第 3 个会话行附近，先列出来）
    var rows = 会话行();
    报告 += "会话行 " + rows.length + " 个\n";
    rows.forEach(function (n, i) {
        var b = n.bounds();
        报告 += "  第" + (i + 1) + "行 " + 心X(b) + "," + 心Y(b) + " " + 宽(b) + "x" + 高(b) + "\n";
    });
    // 孙先生是第 3 个真人会话（袁先生、张秀娟 之后）
    if (rows.length >= 3) {
        rows[2].click();
        return "按行位置点第 3 行";
    }
    return "没找到可点会话行";
}

launchApp("BOSS直聘");
sleep(4500);
报告 += "启动后前台: " + currentPackage() + "\n";

var 已进 = false;
for (var step = 1; step <= 5; step++) {
    if (currentPackage() != PKG) {
        报告 += step + ": 不在 BOSS，重新拉起\n";
        launchApp("BOSS直聘");
        sleep(4000);
        continue;
    }
    if (在聊天页()) {
        报告 += step + ": 已在孙先生聊天页\n";
        已进 = true;
        break;
    }
    if (在消息列表()) {
        报告 += step + ": 消息列表，点孙先生那一栏 → ";
        var r = 点孙先生();
        报告 += r + "\n";
        sleep(2600);
        if (在聊天页()) { 报告 += "  进去了\n"; 已进 = true; break; }
        报告 += "  没进去\n";
        continue;
    }
    var 消息 = 找页签("消息");
    if (消息) {
        报告 += step + ": 首页，点底部消息\n";
        点文字(消息);
        sleep(2400);
        continue;
    }
    报告 += step + ": 认不出界面，back 一层\n";
    back();
    sleep(1600);
    if (currentPackage() != PKG) { launchApp("BOSS直聘"); sleep(4000); }
}

// 顺手把消息列表这页的文字全列出来，供下一版定坐标
if (在消息列表()) {
    var ts = [];
    className("android.widget.TextView").find().forEach(function (n) {
        var s = n.text();
        if (s) ts.push(s);
    });
    报告 += "\n消息列表文字 " + ts.length + " 条: " + ts.join(" / ") + "\n";
}

if (!已进) {
    setClip(报告 + "\n没进到聊天页，脚本停手。");
    toastLog("没进聊天页");
    exit();
}

var 输入 = className("android.widget.EditText").findOne(4000);
var ib = 输入.bounds();
报告 += "输入框: " + ib.left + "," + ib.top + " " + 宽(ib) + "x" + 高(ib) + "\n";

function 面板开了() { return textContains("语音通话").findOnce(900) != null; }
function 发出去了() { return !!textContains("杜全龙").className("android.widget.TextView").findOnce(2000); }

function 右侧候选() {
    var arr = [];
    packageName(PKG).clickable(true).find().forEach(function (n) {
        var b = n.bounds();
        var w = 宽(b), h = 高(b);
        if (w <= 0 || h <= 0 || w > 320 || h > 320) return;
        if (心X(b) > device.width * 0.60 && 心Y(b) > 屏H() * 0.42) arr.push(n);
    });
    return arr;
}

function 列清单(标签) {
    var arr = 右侧候选(), out = [];
    arr.forEach(function (n) {
        var b = n.bounds();
        out.push("[" + (n.text() || n.desc() || "无字") + "@" + 心X(b) + "," + 心Y(b) + " " + 宽(b) + "x" + 高(b) + "]");
    });
    报告 += "\n" + 标签 + "：" + arr.length + " 个\n" + out.join("  ") + "\n";
    return arr;
}

var 空框 = 列清单("空框状态");
报告 += "空框时面板: " + 面板开了() + "\n";

var 命中 = null;
for (var i2 = 0; i2 < 空框.length; i2++) {
    var n2 = 空框[i2], b2 = n2.bounds();
    if (心Y(b2) < 屏H() * 0.6) continue;
    if (心Y(b2) < ib.top - 160) continue;
    报告 += "试 @" + 心X(b2) + "," + 心Y(b2) + " → ";
    n2.click();
    sleep(1200);
    if (面板开了()) { 报告 += "命中，这就是加号\n"; 命中 = n2; break; }
    报告 += "无反应\n";
}

if (命中) {
    var hb = 命中.bounds();
    报告 += "加号: " + 心X(hb) + "," + 心Y(hb) + "\n";
    var arr9 = 右侧候选(), mr = null;
    arr9.forEach(function (n) { if (!mr || 心X(n.bounds()) > 心X(mr.bounds())) mr = n; });
    if (mr) { mr.click(); sleep(1200); }
    报告 += "收面板后: " + 面板开了() + "\n";
}

输入.click();
sleep(900);
try { 输入.setText(话术); } catch (e) { 报告 += "setText 异常: " + e + "\n"; }
sleep(2400);
var nb = 输入.bounds();
报告 += "\n填字后输入框: " + nb.left + "," + nb.top + " " + 宽(nb) + "x" + 高(nb) + "（位移 " + (nb.top - ib.top) + "）\n";
列清单("填字后状态");

var 已发 = false;
if (命中) {
    var 原Y = 心Y(命中.bounds());
    for (var r = 1; r <= 3 && !已发; r++) {
        var arr3 = 右侧候选(), 目标 = null, 最小差 = 99999;
        arr3.forEach(function (n) {
            var b = n.bounds();
            var d = Math.abs(心Y(b) - 原Y);
            if (d < 最小差) { 最小差 = d; 目标 = n; }
        });
        if (!目标) { 报告 += "第" + r + "次无候选\n"; break; }
        var tb = 目标.bounds();
        报告 += "第" + r + "次点 @" + 心X(tb) + "," + 心Y(tb) + "（距加号位 " + 最小差 + "）→ ";
        目标.click();
        sleep(2100);
        已发 = 发出去了();
        报告 += "发出 " + 已发 + "\n";
    }
}

sleep(700);
setClip(报告);
toastLog(已发 ? "发出去了" : "没发出去");
