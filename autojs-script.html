手机自动投递 · 第 19 步

Shizuku 授权后，用系统 input tap 真点。
先装 Shizuku 并启动，回到 AutoJs6 里给它授权，再来跑这个。脚本第一步就自检 shell 权限，没权限会直接喊你去授权。
1

点下面按钮，脚本进剪贴板

2

AutoJs6 新建 19.js，粘贴，保存

3

停在孙先生聊天页，按 Home 退出

4

跑 19.js，看它第一步的自检码

脚本全文（按钮失灵时手动复制）
/**
 * 手机自动投递 · 第 19 步：走 Shizuku 的 input tap，真正点一下屏幕
 */

var 话术 = "孙先生您好，我是杜全龙，18 年 ICT/FCT 测试治具与 FPC 连接器治具机械设计经验，熟练使用 SolidWorks，做过手机 FCT 测试设备、保压盒缓存机构、双车床机器人上下料等项目，服务过华为、OPPO、比亚迪、小米等客户。看到贵司招机械/结构工程师，5-10 年经验、大专，跟我的履历对得上。方便的话我把简历发您看下？";

auto.waitFor();
sleep(800);

var 报告 = "";

// 0) shell 权限自检
var 自检 = null;
try {
    自检 = shell("echo ok", true);
    报告 += "shell 自检: code=" + 自检.code + " out=" + (自检.result || "") + "\n";
} catch (e) {
    报告 += "shell 异常: " + e + "\n";
}
if (!自检 || 自检.code != 0) {
    setClip(报告 + "shell 没权限，先去 Shizuku 里给 AutoJs6 授权，再回来跑这个");
    toastLog("shell 没权限");
    exit();
}

launchApp("BOSS直聘");
sleep(4500);
报告 += "包名: " + currentPackage() + "\n";

var 输入 = className("android.widget.EditText").findOne(6000);
if (!输入) {
    setClip(报告 + "没找到输入框，先进到聊天页");
    toastLog("没找到输入框");
    exit();
}

function 点(X, Y) {
    shell("input tap " + X + " " + Y, true);
}

function 面板出现了() {
    return textContains("语音通话").findOnce(1100) != null ||
           textContains("语音").findOnce(1100) != null;
}

function 发出去了() {
    return !!textContains("杜全龙").className("android.widget.TextView").findOnce(2200);
}

输入.click();
sleep(1200);

var 框 = 输入.bounds();
var X = device.width - 100;
报告 += "输入框: " + 框.left + "," + 框.top + " " + 框.width() + "x" + 框.height() + "\n";
报告 += "竖线 x=" + X + "\n";

var 找到Y = -1, 试过 = 0;
for (var y = 框.top - 80; y <= 框.top + 框.height() + 80; y += 25) {
    点(X, y);
    sleep(1100);
    试过++;
    if (面板出现了()) {
        找到Y = y;
        报告 += "命中加号 @ " + X + "," + y + "  试了 " + 试过 + " 次\n";
        break;
    }
    if (textContains("表情").exists()) { 输入.click(); sleep(800); }
}

if (找到Y < 0) {
    setClip(报告 + "input tap 也点不出加号，试了 " + 试过 + " 个点");
    toastLog("input tap 也点不中");
    exit();
}

点(X, 找到Y);
sleep(1300);
if (面板出现了()) { back(); sleep(1100); 报告 += "面板用返回收起\n"; }
else 报告 += "面板已收起\n";

输入.click();
sleep(800);
try { 输入.setText(话术); } catch (e) { 报告 += "setText 异常: " + e + "\n"; }
sleep(2200);
报告 += "话术已填入\n";

var 已发 = false;
for (var r = 1; r <= 3 && !已发; r++) {
    点(X, 找到Y);
    sleep(2200);
    已发 = 发出去了();
    报告 += "第" + r + "次 input tap 发送: " + 已发 + "\n";
    if (!已发 && 面板出现了()) {
        点(X, 找到Y);
        sleep(1200);
        报告 += "  面板又开了，已收起\n";
    }
}

sleep(1000);
setClip(报告);
toastLog(已发 ? "发出去了" : "还是没发出去");
自检码是 0 才继续往下跑
