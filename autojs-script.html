/**
 * 手机自动投递 · 第 24 步：用节点点击打发送键（修好 bounds 取宽高）
 * v23 报 TypeError: height 是 number 而非函数 —— bounds 的宽高在 AutoJs6 里是属性，
 * 这版一律用 right-left / bottom-top 算，不调方法。
 */

var PKG = "com.hpbr.bosszhipin";
var 话术 = "孙先生您好，我是杜全龙，18 年 ICT/FCT 测试治具与 FPC 连接器治具机械设计经验，熟练使用 SolidWorks，做过手机 FCT 测试设备、保压盒缓存机构、双车床机器人上下料等项目，服务过华为、OPPO、比亚迪、小米等客户。看到贵司招机械/结构工程师，5-10 年经验、大专，跟我的履历对得上。方便的话我把简历发您看下？";

auto.waitFor();
sleep(800);
var 报告 = "";

function 宽(b) { return b.right - b.left; }
function 高(b) { return b.bottom - b.top; }
function 心X(b) { return (b.left + b.right) / 2; }
function 心Y(b) { return (b.top + b.bottom) / 2; }

launchApp("BOSS直聘");
sleep(4200);
报告 += "包名: " + currentPackage() + "\n";

var 输入 = className("android.widget.EditText").findOne(6000);
if (!输入) {
    setClip(报告 + "没找到输入框，先手动进到孙先生聊天页再跑");
    toastLog("没找到输入框");
    exit();
}
var ib = 输入.bounds();
报告 += "输入框: " + ib.left + "," + ib.top + " " + 宽(ib) + "x" + 高(ib) + "\n";

function 找右侧节点() {
    var best = null, bestX = -1;
    packageName(PKG).clickable(true).find().forEach(function (n) {
        var b = n.bounds();
        var w = 宽(b), h = 高(b);
        if (w <= 0 || h <= 0) return;
        var cx = 心X(b), cy = 心Y(b);
        if (cx > device.width * 0.87 && w < 260 && h < 260 &&
            cy > device.height * 0.40 && cy < device.height * 0.80) {
            if (cx > bestX) { bestX = cx; best = b; }
        }
    });
    return best;
}

function 面板开了() { return textContains("语音通话").findOnce(900) != null; }
function 发出去了() { return !!textContains("杜全龙").className("android.widget.TextView").findOnce(2000); }

var kb = 找右侧节点();
if (!kb) {
    setClip(报告 + "聊天页右侧没扫到可点节点");
    toastLog("右侧没节点");
    exit();
}
报告 += "右侧节点 @" + 心X(kb) + "," + 心Y(kb) + " " + 宽(kb) + "x" + 高(kb) + "\n";
报告 += "点前面板: " + 面板开了() + "\n";

// 先收掉已开的面板
if (面板开了()) {
    click(心X(kb), 心Y(kb));
    sleep(1200);
    报告 += "收面板尝试后: " + 面板开了() + "\n";
}

// 实验一：点这个节点，看面板出不出来（证明节点点击对该位置有效）
var 点成功 = false;
text("介绍自己回复率更高哦~").findOnce(800);
var 目标 = textContains("介绍自己").findOnce(600);
if (目标) {
    var q = 目标;
    for (var i = 0; i < 6 && q && !q.clickable(); i++) q = q.parent();
}
// 用节点树里的无文字可点节点点击
var 节点 = null;
packageName(PKG).clickable(true).find().forEach(function (n) {
    var b = n.bounds();
    var w = 宽(b), h = 高(b);
    if (w <= 0 || h <= 0) return;
    if (心X(b) > device.width * 0.87 && w < 260 && h < 260 &&
        心Y(b) > device.height * 0.40 && 心Y(b) < device.height * 0.80) {
        if (!节点 || 心X(b) > 心X(节点.bounds())) 节点 = n;
    }
});
if (节点) {
    节点.click();
    sleep(1300);
    点成功 = 面板开了();
    报告 += "实验：节点 click → 面板出现 " + 点成功 + "\n";
    if (点成功) {
        // 再点收面板
        var 再 = null;
        packageName(PKG).clickable(true).find().forEach(function (n) {
            var b = n.bounds();
            var w = 宽(b), h = 高(b);
            if (w <= 0 || h <= 0) return;
            if (心X(b) > device.width * 0.87 && w < 260 && h < 260 &&
                心Y(b) > device.height * 0.40 && 心Y(b) < device.height * 0.80) {
                if (!再 || 心X(b) > 心X(再.bounds())) 再 = n;
            }
        });
        if (再) { 再.click(); sleep(1300); }
    }
} else {
    报告 += "没拿到可点节点\n";
}
报告 += "收回后还有面板: " + 面板开了() + "\n";

// 填话术
输入.click();
sleep(900);
try { 输入.setText(话术); } catch (e) { 报告 += "setText 异常: " + e + "\n"; }
sleep(2200);
报告 += "话术已填入\n";

// 实验二：填完再点同一位置
var 已发 = false;
for (var r = 1; r <= 3 && !已发; r++) {
    var 位 = 找右侧节点();
    if (!位) { 报告 += "第" + r + "次没扫到右侧节点\n"; break; }
    // 优先节点点击，失败再退化到坐标（坐标大概率被拦，作对照）
    var n2 = null;
    packageName(PKG).clickable(true).find().forEach(function (n) {
        var b = n.bounds();
        var w = 宽(b), h = 高(b);
        if (w <= 0 || h <= 0) return;
        if (心X(b) > device.width * 0.87 && w < 260 && h < 260 &&
            心Y(b) > device.height * 0.40 && 心Y(b) < device.height * 0.80) {
            if (!n2 || 心X(b) > 心X(n2.bounds())) n2 = n;
        }
    });
    if (n2) { n2.click(); } else { click(心X(位), 心Y(位)); }
    sleep(2100);
    已发 = 发出去了();
    报告 += "第" + r + "次点发送位 @" + 心X(位) + "," + 心Y(位) + " → 发出: " + 已发 + "\n";
    if (!已发 && 面板开了()) {
        var 收2 = 找右侧节点();
        if (收2) { click(心X(收2), 心Y(收2)); sleep(1200); }
        报告 += "  面板又开了，已收起\n";
    }
}

sleep(1000);
setClip(报告);
toastLog(已发 ? "发出去了" : "没发出去");
