/**
 * 手机自动投递 · 第 7 步：把定制话术发出去
 * 前提：已经停在 BOSS 的聊天页（输入框可见）
 */

function 扫描() {
    var 全部 = classNameMatches(/.+/).find();
    var 行 = [];
    for (var i = 0; i < 全部.size(); i++) {
        var w = 全部.get(i);
        var 文 = (w.text() || "").trim();
        var 述 = (w.desc() || "").trim();
        if (!文 && !述) continue;
        var b = w.bounds();
        行.push(w.className().replace("android.widget.", "").replace("android.view.", "") +
            " | " + 文 + " | " + 述 + " | " +
            b.left + "," + b.top + " " + b.width() + "x" + b.height());
    }
    return "节点总数: " + 全部.size() + "  含文字: " + 行.length + "\n----\n" + 行.join("\n");
}

var 话术 = "孙先生您好，我是杜全龙，18 年 ICT/FCT 测试治具与 FPC 连接器治具机械设计经验，熟练使用 SolidWorks，做过手机 FCT 测试设备、保压盒缓存机构、双车床机器人上下料等项目，服务过华为、OPPO、比亚迪、小米等客户。看到贵司招机械/结构工程师，5-10 年经验、大专，跟我的履历对得上。方便的话我把简历发您看下？";

var 包 = currentPackage();
if (包 != "com.hpbr.bosszhipin") {
    setClip("不在 BOSS 前台，当前包名: " + 包);
    toastLog("不在 BOSS 前台");
} else {
    var 输入 = className("android.widget.EditText").findOne(6000);
    if (!输入) {
        setClip("没找到输入框，当前界面:\n" + 扫描());
        toastLog("没找到输入框");
    } else {
        toastLog("找到输入框，先点一下聚焦");
        输入.click();
        sleep(1200);

        输入.setText(话术);
        sleep(2000);
        toastLog("文字已填入，找发送按钮");

        var 发送 = text("发送").findOne(2500);
        if (发送) {
            toastLog("找到「发送」按钮，点击");
            发送.click();
        } else {
            toastLog("没有「发送」按钮，按坐标点输入框右侧");
            click(1050, 2291);
        }

        sleep(3000);
        setClip("已尝试发送话术后界面:\n" + 扫描());
        toastLog("发完了，界面结构已复制到剪贴板");
    }
}
