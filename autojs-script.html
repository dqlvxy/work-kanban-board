脚本全文
/**
 * 手机自动投递 · 第 20 步：找「打招呼语」设置入口
 * 思路：BOSS 点「立即沟通」会自动发一条预设招呼语（首投已验证送达），
 * 把那条预设改成杜哥的话术，就等于自动投递，完全不需要点发送键。
 */

auto.waitFor();
sleep(800);
var 报告 = "";

launchApp("BOSS直聘");
sleep(4500);
报告 += "包名: " + currentPackage() + "\n";

function 抓页(标签) {
    var 文 = [], 钮 = [];
    className("android.widget.TextView").find().forEach(function (n) {
        var t = n.text();
        if (t && t.length) 文.push(t);
    });
    var 全 = classNameMatches(/.*/).find();
    全.forEach(function (n) {
        if (!n.clickable()) return;
        var d = (n.desc() || "") + "|" + (n.text() || "");
        var b = n.bounds();
        if (b.width() > 0 && b.height() > 0) 钮.push(d + "@" + (b.left + b.width() / 2) + "," + (b.top + b.height() / 2));
    });
    报告 += "\n===== " + 标签 + " =====\n";
    报告 += "文字 " + 文.length + " 个: " + 文.join(" / ") + "\n";
    报告 += "可点 " + 钮.length + " 个: " + 钮.slice(0, 40).join("  ") + "\n";
}

// 1) 退到首页：底部有「职位」且在最下
var 到家 = false;
for (var i = 0; i < 7; i++) {
    var 职 = text("职位").findOnce(500);
    if (职 && 职.bounds().top > device.height * 0.85) { 到家 = true; break; }
    back();
    sleep(1400);
}
报告 += (到家 ? "已回到首页\n" : "没能确认首页，硬着头皮往下走\n");
抓页("首页");

// 2) 点底部「我的」
var 我的 = text("我的").findOnce(800);
if (我的 && 我的.bounds().top > device.height * 0.8) {
    var p = 我的;
    for (var k = 0; k < 4 && p && !p.clickable(); k++) p = p.parent();
    if (p) { p.click(); sleep(2200); 报告 += "点了「我的」\n"; }
    else { 我的.click(); sleep(2200); 报告 += "点了「我的」（直接）\n"; }
} else {
    报告 += "底部没找到「我的」\n";
}
抓页("我的页");

// 3) 我的页里找「设置」
var 设 = textContains("设置").findOnce(600);
if (设) {
    var q = 设;
    for (var j = 0; j < 4 && q && !q.clickable(); j++) q = q.parent();
    if (q) { q.click(); sleep(2200); 报告 += "点了「设置」\n"; }
}
抓页("设置页");

setClip(报告);
toastLog("页面结构已复制到剪贴板");
