/**
 * 手机自动投递 · 第 26 步：把加号在「空框」和「填好字」两种状态下的节点位置都列出来
 * 依据：v22 在 @1062,2286 84x84 上节点 click 成功弹出了「图片/语音/语音通话」面板，
 * 说明节点点击对加号有效；v24 扫到的是 1413 那个位置，点错了。
 */

var PKG = "com.hpbr.bosszhipin";
var 话术 = "孙先生您好，我是杜全龙，18 年 ICT/FCT 测试治具与 FPC 连接器治具机械设计经验，熟练使用 SolidWorks，做过手机 FCT 测试设备、保压盒缓存机构、双车床机器人上下料等项目，服务过华为、OPPO、比亚迪、小米等客户。看到贵司招机械/结构工程师，5-10 年经验、大专，跟我的履历对得上。方便的话我把简历发您看下？";

auto.waitFor();
sleep(800);
var 报告 = "";

function 宽(b) { return b.right - b.left; }
function 高(b) { return b.bottom - b.top; }
function 心X(b) { return (b.left + b.right) / 2; }
function 心Y(b) { return (b.top + b.bottom) / 2; }

launchApp("BOSS直聘");
sleep(4200);
报告 += "包名: " + currentPackage() + "  屏幕 " + device.width + "x" + device.height + "\n";

var 输入 = className("android.widget.EditText").findOne(6000);
if (!输入) { setClip(报告 + "没找到输入框，先进孙先生聊天页"); toastLog("没输入框"); exit(); }
var ib = 输入.bounds();
报告 += "输入框: " + ib.left + "," + ib.top + " " + 宽(ib) + "x" + 高(ib) + "\n";

function 面板开了() { return textContains("语音通话").findOnce(900) != null; }
function 发出去了() { return !!textContains("杜全龙").className("android.widget.TextView").findOnce(2000); }

// 先把右侧所有小方块列出来
function 右侧候选() {
    var arr = [];
    packageName(PKG).clickable(true).find().forEach(function (n) {
        var b = n.bounds();
        var w = 宽(b), h = 高(b);
        if (w <= 0 || h <= 0) return;
        if (心X(b) > device.width * 0.85 && w < 280 && h < 280) arr.push(n);
    });
    return arr;
}

function 列清单(标签) {
    var arr = 右侧候选(), out = [];
    arr.forEach(function (n) {
        var b = n.bounds();
        out.push((n.text() || n.desc() || "无字") + "@" + 心X(b) + "," + 心Y(b) + " " + 宽(b) + "x" + 高(b));
    });
    报告 += "\n" + 标签 + "：右侧候选 " + arr.length + " 个\n" + out.join("  ") + "\n";
    return arr;
}

// 空框状态清单
var 空框 = 列清单("空框状态");
报告 += "空框时面板: " + 面板开了() + "\n";

// 挑最靠右那个点一下，看能不能弹面板
var 最右 = null;
空框.forEach(function (n) { if (!最右 || 心X(n.bounds()) > 心X(最右.bounds())) 最右 = n; });
if (最右) {
    var rb = 最右.bounds();
    报告 += "试最右 @ " + 心X(rb) + "," + 心Y(rb) + " → ";
    最右.click();
    sleep(1300);
    var 开 = 面板开了();
    报告 += "面板 " + 开 + "\n";
    if (开) {
        // 收掉面板
        var now = 右侧候选(), mr = null;
        now.forEach(function (n) { if (!mr || 心X(n.bounds()) > 心X(mr.bounds())) mr = n; });
        if (mr) { mr.click(); sleep(1300); }
        报告 += "收面板后: " + 面板开了() + "\n";
    }
}

// 填话术
输入.click();
sleep(900);
try { 输入.setText(话术); } catch (e) { 报告 += "setText 异常: " + e + "\n"; }
sleep(2400);
var 新框 = 输入.bounds();
报告 += "填字后输入框: " + 新框.left + "," + 新框.top + " " + 宽(新框) + "x" + 高(新框) + "\n";

// 填字后清单
var 填后 = 列清单("填字后状态");

var 已发 = false;
for (var r = 1; r <= 3 && !已发; r++) {
    var arr = 右侧候选();
    var mr = null;
    arr.forEach(function (n) { if (!mr || 心X(n.bounds()) > 心X(mr.bounds())) mr = n; });
    if (!mr) { 报告 += "第" + r + "次无候选\n"; break; }
    var b2 = mr.bounds();
    报告 += "第" + r + "次点最右 @" + 心X(b2) + "," + 心Y(b2) + " → ";
    mr.click();
    sleep(2100);
    已发 = 发出去了();
    报告 += "发出 " + 已发 + "\n";
    if (!已发 && 面板开了()) { 报告 += "  （面板开出来了）\n"; }
}

sleep(800);
setClip(报告);
toastLog(已发 ? "发出去了" : "没发出去");
