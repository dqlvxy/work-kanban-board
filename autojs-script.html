脚本全文（按钮失灵时手动复制）
/**
 * 手机自动投递 · 第 13 步：不读字数，只看有没有发出新气泡
 * BOSS 输入框是无障碍读不出文本的 Web 控件，text() 恒为空，不能用字数判断成败。
 */

var 话术 = "孙先生您好，我是杜全龙，18 年 ICT/FCT 测试治具与 FPC 连接器治具机械设计经验，熟练使用 SolidWorks，做过手机 FCT 测试设备、保压盒缓存机构、双车床机器人上下料等项目，服务过华为、OPPO、比亚迪、小米等客户。看到贵司招机械/结构工程师，5-10 年经验、大专，跟我的履历对得上。方便的话我把简历发您看下？";

auto.waitFor();
sleep(800);

var 报告 = "";
launchApp("BOSS直聘");
sleep(4500);

if (currentPackage() != "com.hpbr.bosszhipin") {
    setClip("不在 BOSS 前台");
    toastLog("不在 BOSS");
    exit();
}

var 输入 = className("android.widget.EditText").findOne(6000);
if (!输入) {
    setClip("没找到输入框");
    toastLog("没找到输入框");
    exit();
}

// 成功判据：聊天区出现一段含「杜全龙」的 TextView（气泡才会变成 TextView）
function 发出去了() {
    var t = textContains("杜全龙").className("android.widget.TextView").findOnce(2500);
    return !!t;
}

var 已发 = 发出去了();
报告 += "开跑前已有气泡: " + 已发 + "\n";

// 填字（读不到也无所谓，照样填一遍）
输入.click();
sleep(1200);
try { 输入.setText(话术); } catch (e) { 报告 += "setText 异常: " + e + "\n"; }
sleep(1800);

var 框 = 输入.bounds();
var 目标X = device.width - 100;
var 目标Y = 框.centerY();
报告 += "输入框: " + 框.left + "," + 框.top + " " + 框.width() + "x" + 框.height() + "\n";
报告 += "目标点: " + 目标X + "," + 目标Y + "\n";

// 点纸飞机
for (var r = 1; r <= 3 && !已发; r++) {
    click(目标X, 目标Y);
    sleep(2000);
    已发 = 发出去了();
    报告 += "第" + r + "次 click: " + 已发 + "\n";
    if (已发) break;

    try { swipe(目标X - 4, 目标Y - 4, 目标X + 4, 目标Y + 4, 80); } catch (e) {}
    sleep(2000);
    已发 = 发出去了();
    报告 += "第" + r + "次 swipe: " + 已发 + "\n";
}

// 手势自检：点笑脸
if (!已发) {
    var 笑X = device.width - 282;
    var 笑Y = 框.centerY();
    var 前节点 = classNameMatches(/.+/).find().size();
    click(笑X, 笑Y);
    sleep(1800);
    var 后节点 = classNameMatches(/.+/).find().size();
    报告 += "自检 点笑脸: 节点 " + 前节点 + "→" + 后节点 + "\n";
    if (后节点 > 前节点 + 10) 报告 += "结论: 手势能传进 App\n";
    else 报告 += "结论: 手势被忽略\n";
    输入.click();
    sleep(600);
}

sleep(800);
setClip(报告);
toastLog(已发 ? "发出去了" : "没发出去");
发出去了右边会从纸飞机变回加号
