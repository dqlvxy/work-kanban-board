/**
 * 手机自动投递 · 第 25 步：点底部「我的」，摸设置里的打招呼语
 * 依据：v22 报告里四个页签文字齐全（职位/发现/消息/我的），但外层可点节点没文字；
 * v22 误点到的是聊天页键盘节点。这版先确保离开聊天页，再按文字节点向上找可点父节点。
 */

var PKG = "com.hpbr.bosszhipin";
auto.waitFor();
sleep(800);
var 报告 = "";

function 宽(b) { return b.right - b.left; }
function 高(b) { return b.bottom - b.top; }
function 心X(b) { return (b.left + b.right) / 2; }
function 心Y(b) { return (b.top + b.bottom) / 2; }
function 在BOSS() { return currentPackage() === PKG; }

function 回BOSS() {
    for (var i = 0; i < 3 && !在BOSS(); i++) { app.launchPackage(PKG); sleep(2600); }
    return 在BOSS();
}

function 抓页(标签) {
    if (!在BOSS()) { 报告 += "\n===== " + 标签 + " =====\n（不在 BOSS）\n"; return; }
    var 文 = [], 钮 = [];
    packageName(PKG).className("android.widget.TextView").find().forEach(function (n) {
        var t = n.text();
        if (t && t.length) 文.push(t);
    });
    packageName(PKG).find().forEach(function (n) {
        if (!n.clickable()) return;
        var b = n.bounds();
        if (宽(b) > 0 && 高(b) > 0) 钮.push((n.desc() || "") + "|" + (n.text() || "") + "@" + 心X(b) + "," + 心Y(b));
    });
    报告 += "\n===== " + 标签 + " =====\n";
    报告 += "文字 " + 文.length + ": " + 文.join(" / ") + "\n";
    报告 += "可点 " + 钮.length + ": " + 钮.slice(0, 40).join("  ") + "\n";
}

回BOSS();
报告 += "启动后前台: " + currentPackage() + "\n";
sleep(1600);

// 若还停在聊天页（有输入框），先退出来
for (var i = 0; i < 3 && className("android.widget.EditText").exists(); i++) {
    报告 += "还在聊天页，按返回\n";
    back();
    sleep(1600);
    回BOSS();
}
报告 += "输入框还在: " + className("android.widget.EditText").exists() + "\n";

// 找底部四页签里「我的」的文字节点
var 我的 = null;
for (var t = 0; t < 4; t++) {
    var d = text("我的").findOnce(500);
    if (d && d.bounds().top > device.height * 0.85) { 我的 = d; break; }
    if (!d) { back(); sleep(1500); 回BOSS(); }
}
if (!我的) {
    抓页("当前位置");
    setClip(报告 + "\n没在底部找到「我的」");
    toastLog("没找到我的");
    exit();
}
var mb = 我的.bounds();
报告 += "「我的」文字 @" + 心X(mb) + "," + 心Y(mb) + "\n";

// 向上找可点父节点
var q = 我的, 层 = 0;
while (q && !q.clickable() && 层 < 8) { q = q.parent(); 层++; }
报告 += "向上找了 " + 层 + " 层，祖先可点: " + (q ? q.clickable() : false) + "\n";
if (!q) q = 我的;
q.click();
sleep(2600);
报告 += "点完前台: " + currentPackage() + "\n";
抓页("点我的之后");

// 在页面里找设置/打招呼
var 候选 = ["设置", "打招呼", "隐私"];
for (var c = 0; c < 候选.length; c++) {
    var n = textContains(候选[c]).findOnce(500);
    if (n) {
        var p = n, k = 0;
        while (p && !p.clickable() && k < 8) { p = p.parent(); k++; }
        if (!p) p = n;
        p.click();
        sleep(2400);
        报告 += "点了「" + 候选[c] + "」\n";
        抓页("点" + 候选[c] + "之后");
        break;
    }
}

setClip(报告);
toastLog("结构已复制到剪贴板");
底部页签是原生控件，点得动
