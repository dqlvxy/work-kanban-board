/**
 * 手机自动投递 · 第 33 步：滚一屏读一屏，读到孙先生再点
 * 用户口径：逐步往下滑、逐步识别，别每行点进去再退出，太费时间。
 * 滚动用节点 scrollForward（无障碍动作），不用坐标 swipe（鸿蒙拦手势）。
 */

var PKG = "com.hpbr.bosszhipin";
var 话术 = "孙先生您好，我是杜全龙，18 年 ICT/FCT 测试治具与 FPC 连接器治具机械设计经验，熟练使用 SolidWorks，做过手机 FCT 测试设备、保压盒缓存机构、双车床机器人上下料等项目，服务过华为、OPPO、比亚迪、小米等客户。看到贵司招机械/结构工程师，5-10 年经验、大专，跟我的履历对得上。方便的话我把简历发您看下？";

auto.waitFor();
sleep(600);
var 报告 = "";

function 宽(b) { return b.right - b.left; }
function 高(b) { return b.bottom - b.top; }
function 心X(b) { return (b.left + b.right) / 2; }
function 心Y(b) { return (b.top + b.bottom) / 2; }
function 屏H() { return device.height; }

launchApp("BOSS直聘");
sleep(4500);
if (currentPackage() != PKG) { launchApp("BOSS直聘"); sleep(4000); }
报告 += "前台 " + currentPackage() + "\n";
if (currentPackage() != PKG) { setClip(报告 + "BOSS 没到前台"); toastLog("停手"); exit(); }
sleep(1000);

function 在聊天页() {
    var e = className("android.widget.EditText").findOnce();
    return e != null && e.bounds().top > 屏H() * 0.70;
}

// 滚一屏：优先找可滚动容器，用无障碍滚动动作
function 滚动一屏() {
    var s = classNameMatches(/.*/).scrollable(true).findOnce(1000);
    if (!s) s = className("androidx.recyclerview.widget.RecyclerView").findOnce(800);
    if (!s) s = className("android.widget.ListView").findOnce(600);
    if (s) {
        try { s.scrollForward(); return "节点滚动"; } catch (e) { }
        try { s.scrollForward(); return "节点滚动2"; } catch (e2) { }
    }
    // 退路：手势滑动（鸿蒙大概率拦，但试一次）
    try {
        var h = 屏H();
        swipe(device.width / 2, h * 0.75, device.width / 2, h * 0.30, 400);
        return "手势滑动";
    } catch (e3) { }
    return "";
}

// 当前屏读到的文字
function 读文字() {
    var arr = [];
    className("android.widget.TextView").find().forEach(function (n) {
        var t = n.text();
        if (t) arr.push(t);
    });
    return arr;
}

// 找到孙先生所在的可点行容器
function 孙先生所在行(节点) {
    var y = 心Y(节点.bounds());
    var 行 = null;
    className("android.view.ViewGroup").find().forEach(function (n) {
        var b = n.bounds();
        if (宽(b) > 1000 && 高(b) > 60 && 高(b) < 340 && Math.abs(心Y(b) - y) < 160) 行 = n;
    });
    return 行;
}

// ———— 滚屏找孙先生 ————
var 找到 = 在聊天页();
if (在聊天页()) 报告 += "已经在某个聊天页\n";

if (!找到) {
    for (var k = 1; k <= 8; k++) {
        var 孙 = textContains("孙先生").findOnce(900);
        var 字 = 读文字();
        报告 += "第" + k + "屏：文字 " + 字.length + " 条" + (孙 ? "，读到孙先生 " + Math.round(心Y(孙.bounds())) : "，没孙先生") + "\n";
        if (孙) {
            var 行 = 孙先生所在行(孙);
            报告 += "  点这一行…";
            if (行 && 行.clickable()) 行.click(); else if (行) 行.click(); else 孙.click();
            sleep(2700);
            if (在聊天页()) { 报告 += "进聊天页了\n"; 找到 = true; }
            else 报告 += "没进去\n";
            break;
        }
        var 方式 = 滚动一屏();
        if (!方式) { 报告 += "  滚不动，停\n"; break; }
        报告 += "  滚动方式: " + 方式 + "\n";
        sleep(1700);
    }
}

if (!找到) {
    // 把最后读到的文字全带上，便于诊断
    var 尾 = 读文字();
    setClip(报告 + "\n最后屏文字: " + 尾.join(" / "));
    toastLog("没找到孙先生");
    exit();
}

// ———— 干活 ————
var 输入 = className("android.widget.EditText").findOne(4000);
if (!输入) { setClip(报告 + "\n聊天页没输入框"); toastLog("没输入框"); exit(); }
var ib = 输入.bounds();
报告 += "输入框: " + ib.left + "," + ib.top + " " + 宽(ib) + "x" + 高(ib) + "\n";

function 面板开了() { return textContains("语音通话").findOnce(900) != null; }
function 发出去了() { return !!textContains("杜全龙").className("android.widget.TextView").findOnce(2000); }

function 右侧候选() {
    var arr = [];
    classNameMatches(/.*/).find().forEach(function (n) {
        try {
            var b = n.bounds();
            var w = 宽(b), h = 高(b);
            if (w <= 0 || h <= 0 || w > 320 || h > 320) return;
            if (心X(b) > device.width * 0.60 && 心Y(b) > 屏H() * 0.42) arr.push(n);
        } catch (e) { }
    });
    return arr;
}

function 列清单(标签) {
    var arr = 右侧候选(), out = [];
    arr.forEach(function (n) {
        var b = n.bounds();
        out.push("[" + (n.text() || n.desc() || "无字") + "@" + Math.round(心X(b)) + "," + Math.round(心Y(b)) + " " + 宽(b) + "x" + 高(b) + "]");
    });
    报告 += "\n" + 标签 + "：" + arr.length + " 个\n" + out.join("  ") + "\n";
    return arr;
}

var 空框 = 列清单("空框状态");
报告 += "空框时面板: " + 面板开了() + "\n";

var 命中 = null;
for (var j = 0; j < 空框.length; j++) {
    var n2 = 空框[j], b2 = n2.bounds();
    if (心Y(b2) < 屏H() * 0.60) continue;
    if (心Y(b2) < ib.top - 160) continue;
    报告 += "试 @" + Math.round(心X(b2)) + "," + Math.round(心Y(b2)) + " → ";
    n2.click();
    sleep(1200);
    if (面板开了()) { 报告 += "命中，这就是加号\n"; 命中 = n2; break; }
    报告 += "无反应\n";
}

if (命中) {
    var hb = 命中.bounds();
    报告 += "加号: " + Math.round(心X(hb)) + "," + Math.round(心Y(hb)) + "\n";
    var a9 = 右侧候选(), mr = null;
    a9.forEach(function (n) { if (!mr || 心X(n.bounds()) > 心X(mr.bounds())) mr = n; });
    if (mr) { mr.click(); sleep(1200); }
    报告 += "收面板后: " + 面板开了() + "\n";
}

输入.click();
sleep(900);
try { 输入.setText(话术); } catch (e) { 报告 += "setText 异常: " + e + "\n"; }
sleep(2400);
var nb = 输入.bounds();
报告 += "\n填字后输入框: " + nb.left + "," + nb.top + "（位移 " + (nb.top - ib.top) + "）\n";
列清单("填字后状态");

var 已发 = false;
if (命中) {
    var 原Y = 心Y(命中.bounds());
    for (var r = 1; r <= 3 && !已发; r++) {
        var a3 = 右侧候选(), 目标 = null, 最小差 = 99999;
        a3.forEach(function (n) {
            var b = n.bounds();
            var d = Math.abs(心Y(b) - 原Y);
            if (d < 最小差) { 最小差 = d; 目标 = n; }
        });
        if (!目标) { 报告 += "第" + r + "次无候选\n"; break; }
        var tb = 目标.bounds();
        报告 += "第" + r + "次点 @" + Math.round(心X(tb)) + "," + Math.round(心Y(tb)) + " → ";
        目标.click();
        sleep(2100);
        已发 = 发出去了();
        报告 += "发出 " + 已发 + "\n";
    }
}

sleep(600);
setClip(报告);
toastLog(已发 ? "发出去了" : "没发出去");
