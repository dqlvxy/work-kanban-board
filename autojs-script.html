/**
 * 手机自动投递 · 第 27 步：先校验在聊天页，再列空框/填字两态节点
 * v26 教训：脚本跑时人停在简历页，输入框 y=1353，聊天页应在 y≈2255，
 * 按同一套条件扫右侧节点，点到了右上角分享键，弹出在线简历预览。
 */

var PKG = "com.hpbr.bosszhipin";
var 话术 = "孙先生您好，我是杜全龙，18 年 ICT/FCT 测试治具与 FPC 连接器治具机械设计经验，熟练使用 SolidWorks，做过手机 FCT 测试设备、保压盒缓存机构、双车床机器人上下料等项目，服务过华为、OPPO、比亚迪、小米等客户。看到贵司招机械/结构工程师，5-10 年经验、大专，跟我的履历对得上。方便的话我把简历发您看下？";

auto.waitFor();
sleep(800);
var 报告 = "";

function 宽(b) { return b.right - b.left; }
function 高(b) { return b.bottom - b.top; }
function 心X(b) { return (b.left + b.right) / 2; }
function 心Y(b) { return (b.top + b.bottom) / 2; }

报告 += "包名: " + currentPackage() + "  屏幕 " + device.width + "x" + device.height + "\n";

var 输入 = className("android.widget.EditText").findOne(5000);
if (!输入) {
    setClip(报告 + "\n没有输入框 —— 请先进到孙先生聊天页再跑");
    toastLog("不在聊天页");
    exit();
}
var ib = 输入.bounds();
报告 += "输入框: " + ib.left + "," + ib.top + " " + 宽(ib) + "x" + 高(ib) + "\n";

// 聊天页判据：输入框在屏幕下部（y 起点 > 70% 屏高）
if (ib.top < device.height * 0.70) {
    setClip(报告 + "\n输入框 top=" + ib.top + "，不到屏高 70%（" + Math.round(device.height * 0.7) + "）\n" +
        "当前页不是孙先生聊天页，脚本已停手，免得乱点。\n请手动进到聊天页再跑。");
    toastLog("不在聊天页，已停手");
    exit();
}

function 面板开了() { return textContains("语音通话").findOnce(900) != null; }
function 发出去了() { return !!textContains("杜全龙").className("android.widget.TextView").findOnce(2000); }

// 放宽条件：右侧区域所有可点小方块
function 右侧候选() {
    var arr = [];
    packageName(PKG).clickable(true).find().forEach(function (n) {
        var b = n.bounds();
        var w = 宽(b), h = 高(b);
        if (w <= 0 || h <= 0 || w > 320 || h > 320) return;
        if (心X(b) > device.width * 0.60 && 心Y(b) > device.height * 0.42) arr.push(n);
    });
    return arr;
}

function 列清单(标签) {
    var arr = 右侧候选(), out = [];
    arr.forEach(function (n) {
        var b = n.bounds();
        out.push("[" + (n.text() || n.desc() || "无字") + "@" + 心X(b) + "," + 心Y(b) + " " + 宽(b) + "x" + 高(b) + "]");
    });
    报告 += "\n" + 标签 + "：" + arr.length + " 个\n" + out.join("  ") + "\n";
    return arr;
}

var 空框 = 列清单("空框状态");
报告 += "空框时面板: " + 面板开了() + "\n";

// 逐个试空框态的候选，找哪个能弹出面板（找加号真身）
var 命中 = null;
for (var i = 0; i < 空框.length; i++) {
    var n = 空框[i], b = n.bounds();
    if (心Y(b) < device.height * 0.6) continue;   // 上部的不试，省得点出别的页
    if (心Y(b) < ib.top - 120) continue;          // 输入框上方太远的也不试
    报告 += "试 @" + 心X(b) + "," + 心Y(b) + " → ";
    n.click();
    sleep(1200);
    if (面板开了()) { 报告 += "命中！\n"; 命中 = n; break; }
    报告 += "无反应\n";
}

if (命中) {
    var hb = 命中.bounds();
    报告 += "加号真身: " + 心X(hb) + "," + 心Y(hb) + " " + 宽(hb) + "x" + 高(hb) + "\n";
    // 收面板
    var arr2 = 右侧候选(), mr = null;
    arr2.forEach(function (n) { if (!mr || 心X(n.bounds()) > 心X(mr.bounds())) mr = n; });
    if (mr) { mr.click(); sleep(1200); }
    报告 += "收面板后: " + 面板开了() + "\n";
} else {
    报告 += "空框态没试出加号\n";
}

// 填话术
输入.click();
sleep(900);
try { 输入.setText(话术); } catch (e) { 报告 += "setText 异常: " + e + "\n"; }
sleep(2400);
var nb = 输入.bounds();
报告 += "\n填字后输入框: " + nb.left + "," + nb.top + " " + 宽(nb) + "x" + 高(nb) + "（相对空框位移 " + (nb.top - ib.top) + "）\n";
列清单("填字后状态");

var 已发 = false;
if (命中) {
    var 原Y = 心Y(命中.bounds());
    for (var r = 1; r <= 3 && !已发; r++) {
        var arr3 = 右侧候选(), 目标 = null, 最小差 = 99999;
        arr3.forEach(function (n) {
            var b = n.bounds();
            var d = Math.abs(心Y(b) - 原Y);
            if (d < 最小差) { 最小差 = d; 目标 = n; }
        });
        if (!目标) { 报告 += "第" + r + "次无候选\n"; break; }
        var tb = 目标.bounds();
        报告 += "第" + r + "次点 @" + 心X(tb) + "," + 心Y(tb) + "（与加号位差 " + 最小差 + "）→ ";
        目标.click();
        sleep(2100);
        已发 = 发出去了();
        报告 += "发出 " + 已发 + "\n";
        if (!已发 && 面板开了()) 报告 += "  （又把面板点开了）\n";
    }
}

sleep(800);
setClip(报告);
toastLog(已发 ? "发出去了" : "没发出去");
这版会挨个试空框态的候选，找出哪个是加号
