脚本全文（按钮失灵时手动复制）
/**
 * 手机自动投递 · 第 9 步：按界面结构找发送按钮
 * 不再硬点坐标。筛出右下区域所有可点击节点，逐个试，发出去就停。
 */

function 扫描() {
    var 全部 = classNameMatches(/.+/).find();
    var 行 = [];
    for (var i = 0; i < 全部.size(); i++) {
        var w = 全部.get(i);
        var 文 = (w.text() || "").trim();
        var 述 = (w.desc() || "").trim();
        if (!文 && !述) continue;
        var b = w.bounds();
        行.push(w.className().replace("android.widget.", "").replace("android.view.", "") +
            " | " + 文 + " | " + 述 + " | " +
            b.left + "," + b.top + " " + b.width() + "x" + b.height());
    }
    return 行.join("\n");
}

    var 输入 = className("android.widget.EditText").findOne(6000);
    if (!输入) {
        setClip(报告 + "没找到输入框\n" + 扫描());
        toastLog("没找到输入框");
    } else {
        // 点一下输入框，让键盘弹起来，布局才是真实发送时的样子
        输入.click();
        sleep(1500);

        var 原有 = (输入.text() || "").trim();
        报告 += "输入框字数: " + 原有.length + "\n";
        if (原有.length < 10) {
            setClip(报告 + "输入框是空的，先把 7.js 跑一遍填字，再来跑这个");
            toastLog("输入框是空的");
        } else {
            var 框 = 输入.bounds();
            报告 += "输入框位置: " + 框.left + "," + 框.top + " " + 框.width() + "x" + 框.height() + "\n";

            // 筛右下区域可点击节点
            var 候选 = [];
            var 全部 = classNameMatches(/.+/).find();
            for (var i = 0; i < 全部.size(); i++) {
                var w = 全部.get(i);
                var b = w.bounds();
                if (b.width() <= 0 || b.height() <= 0) continue;
                if (!w.clickable()) continue;
                if (b.centerX() < device.width * 0.85) continue;
                if (b.centerY() < 框.top - 150 || b.centerY() > 框.top + 框.height() + 150) continue;
                候选.push({ 节点: w, x: b.centerX(), y: b.centerY(), 类: w.className(), 述: w.desc() || "" });
            }

            报告 += "候选按钮 " + 候选.length + " 个\n";
            for (var j = 0; j < 候选.length; j++) {
                报告 += "  候选" + j + ": " + 候选[j].类 + " | " + 候选[j].述 +
                    " | " + 候选[j].x + "," + 候选[j].y + "\n";
            }

            var 已发 = false;
            for (var k = 0; k < 候选.length; k++) {
                toastLog("试点候选 " + k + " @" + 候选[k].x + "," + 候选[k].y);
                候选[k].节点.click();
                sleep(1800);
                if (((输入.text() || "").trim()).length < 8) {
                    已发 = true;
                    报告 += "第 " + k + " 个候选点中了，输入框已清空\n";
                    break;
                }
            }

            if (!已发) {
                报告 += "所有候选都没点中，改试三个纵向坐标\n";
                for (var m = -60; m <= 60; m += 60) {
                    click(device.width - 105, Math.round(框.centerY() + m));
                    sleep(1500);
                    if (((输入.text() || "").trim()).length < 8) {
                        已发 = true;
                        报告 += "坐标 " + (框.centerY() + m) + " 点中了\n";
                        break;
                    }
                }
            }

            if (!已发) 报告 += "全部失败，输入框仍有字\n";
            sleep(1500);
            setClip(报告 + "----\n" + 扫描());
            toastLog(已发 ? "发出去了" : "还是没发出去");
        }
    }
