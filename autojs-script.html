/**
 * 手机自动投递 · 第 8 步：补上「发送」这一下
 * 输入框已有文字就不重复填，只负责把发送按钮点准。
 */

function 扫描() {
    var 全部 = classNameMatches(/.+/).find();
    var 行 = [];
    for (var i = 0; i < 全部.size(); i++) {
        var w = 全部.get(i);
        var 文 = (w.text() || "").trim();
        var 述 = (w.desc() || "").trim();
        if (!文 && !述) continue;
        var b = w.bounds();
        行.push(w.className().replace("android.widget.", "").replace("android.view.", "") +
            " | " + 文 + " | " + 述 + " | " +
            b.left + "," + b.top + " " + b.width() + "x" + b.height());
    }
    return "节点总数: " + 全部.size() + "  含文字: " + 行.length + "\n----\n" + 行.join("\n");
}

var 话术 = "孙先生您好，我是杜全龙，18 年 ICT/FCT 测试治具与 FPC 连接器治具机械设计经验，熟练使用 SolidWorks，做过手机 FCT 测试设备、保压盒缓存机构、双车床机器人上下料等项目，服务过华为、OPPO、比亚迪、小米等客户。看到贵司招机械/结构工程师，5-10 年经验、大专，跟我的履历对得上。方便的话我把简历发您看下？";

var 包 = currentPackage();
if (包 != "com.hpbr.bosszhipin") {
    setClip("不在 BOSS 前台，当前包名: " + 包);
    toastLog("不在 BOSS");
} else {
    var 输入 = className("android.widget.EditText").findOne(6000);
    if (!输入) {
        setClip("没找到输入框:\n" + 扫描());
        toastLog("没找到输入框");
    } else {
        var 现有 = (输入.text() || "").trim();
        toastLog("输入框现有字数: " + 现有.length);

        if (现有.length < 20) {
            输入.click();
            sleep(1000);
            输入.setText(话术);
            sleep(1800);
            toastLog("已填入话术");
        } else {
            toastLog("已有内容，不重复填");
            输入.click();
            sleep(800);
        }

        // 优先按文字找发送按钮
        var 发送 = text("发送").findOnce();
        if (!发送) 发送 = descContains("发送").findOnce();
        if (!发送) 发送 = textContains("发送").findOnce();

        if (发送) {
            toastLog("按文字找到发送按钮，点击");
            发送.click();
        } else {
            var b = 输入.bounds();
            var sx = device.width - 105;
            var sy = Math.round(b.top + b.height() * 0.72);
            toastLog("没找到文字按钮，点坐标 " + sx + "," + sy);
            click(sx, sy);
            sleep(1200);
            // 再兜一次：换成输入框中心同高
            if (输入.text() && 输入.text().length > 20) {
                toastLog("第一次没发出去，改点另一高度");
                click(sx, b.centerY());
            }
        }

        sleep(3500);
        setClip("发送后界面:\n输入框剩字: " + ((输入.text() || "").length) + "\n" + 扫描());
        toastLog("结束，结果已复制到剪贴板");
    }
}
