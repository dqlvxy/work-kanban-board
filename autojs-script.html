脚本全文（按钮失灵时手动复制）
/**
 * 手机自动投递 · 第 15 步：改用回车键发送
 * 坐标手势被系统拦了，试试键盘回车通道。
 */

var 话术 = "孙先生您好，我是杜全龙，18 年 ICT/FCT 测试治具与 FPC 连接器治具机械设计经验，熟练使用 SolidWorks，做过手机 FCT 测试设备、保压盒缓存机构、双车床机器人上下料等项目，服务过华为、OPPO、比亚迪、小米等客户。看到贵司招机械/结构工程师，5-10 年经验、大专，跟我的履历对得上。方便的话我把简历发您看下？";

auto.waitFor();
sleep(800);

var 报告 = "";
launchApp("BOSS直聘");
sleep(4500);

if (currentPackage() != "com.hpbr.bosszhipin") {
    setClip("不在 BOSS 前台");
    toastLog("不在 BOSS");
    exit();
}

var 输入 = className("android.widget.EditText").findOne(6000);
if (!输入) {
    setClip("没找到输入框");
    toastLog("没找到输入框");
    exit();
}

function 发出去了() {
    return !!textContains("杜全龙").className("android.widget.TextView").findOnce(2500);
}

输入.click();
sleep(900);
try { 输入.setText(话术); } catch (e) { 报告 += "setText 异常: " + e + "\n"; }
sleep(2200);
报告 += "话术已填入\n";

var 已发 = false;

// 1) press enter
try {
    press("enter");
    sleep(2400);
    已发 = 发出去了();
    报告 += "press enter: " + 已发 + "\n";
} catch (e) {
    报告 += "press enter 异常: " + e + "\n";
}

// 2) keyCode 66
if (!已发) {
    try {
        keyCode(66);
        sleep(2400);
        已发 = 发出去了();
        报告 += "keyCode 66: " + 已发 + "\n";
    } catch (e) { 报告 += "keyCode 异常: " + e + "\n"; }
}

// 3) 输入框 performAction 焦点 + IME 回车（Android 11+）
if (!已发) {
    try {
        输入.performAction(1);
        sleep(500);
        输入.performAction(2097152);
        sleep(2400);
        已发 = 发出去了();
        报告 += "IME ACTION: " + 已发 + "\n";
    } catch (e) { 报告 += "IME 异常: " + e + "\n"; }
}

// 4) shell keyevent（有 root 才可用）
if (!已发) {
    try {
        var r = shell("input keyevent 66", true);
        报告 += "shell keyevent: " + r.code + " " + r.result + "\n";
        sleep(2400);
        已发 = 发出去了();
        报告 += "shell 后: " + 已发 + "\n";
    } catch (e) { 报告 += "shell 异常: " + e + "\n"; }
}

sleep(1000);
setClip(报告);
toastLog(已发 ? "发出去了" : "回车也没发出去");
报告会逐条写四种通道成没成
