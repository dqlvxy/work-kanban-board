/**
 * 手机自动投递 · 第 23 步：用节点点击打发送键
 * 依据：v22 报告里聊天页右侧 1062,1384 是个 clickable 节点（加号）。
 * 坐标手势被拦，但节点 click() 有效（「立即沟通」就是这么点成的）。
 * 这版全程不用坐标，只用节点增删判断界面变化。
 */

var PKG = "com.hpbr.bosszhipin";
var 话术 = "孙先生您好，我是杜全龙，18 年 ICT/FCT 测试治具与 FPC 连接器治具机械设计经验，熟练使用 SolidWorks，做过手机 FCT 测试设备、保压盒缓存机构、双车床机器人上下料等项目，服务过华为、OPPO、比亚迪、小米等客户。看到贵司招机械/结构工程师，5-10 年经验、大专，跟我的履历对得上。方便的话我把简历发您看下？";

auto.waitFor();
sleep(800);
var 报告 = "";

launchApp("BOSS直聘");
sleep(4200);
报告 += "包名: " + currentPackage() + "\n";

var 输入 = className("android.widget.EditText").findOne(6000);
if (!输入) {
    setClip(报告 + "没找到输入框，先手动进到孙先生聊天页再跑");
    toastLog("没找到输入框");
    exit();
}
报告 += "输入框: " + 输入.bounds() + "\n";

// 聊天页右侧那个可点小方块（空框时是加号，有字时是纸飞机）
function 找右侧节点() {
    var best = null;
    packageName(PKG).clickable(true).find().forEach(function (n) {
        var b = n.bounds();
        var cx = b.left + b.width() / 2, cy = b.top + b.height() / 2;
        if (cx > device.width * 0.87 && b.width() < 260 && b.height() < 260 &&
            cy > device.height() * 0.40 && cy < device.height() * 0.78) {
            if (!best || cx > (best.bounds().left + best.bounds().width() / 2)) best = n;
        }
    });
    return best;
}

function 面板开了() {
    return textContains("语音通话").findOnce(900) != null;
}

function 发出去了() {
    return !!textContains("杜全龙").className("android.widget.TextView").findOnce(2000);
}

var 键 = 找右侧节点();
if (!键) {
    setClip(报告 + "聊天页右侧没扫到可点节点");
    toastLog("右侧没节点");
    exit();
}
var kb = 键.bounds();
报告 += "右侧节点 @" + (kb.left + kb.width() / 2) + "," + (kb.top + kb.height() / 2) + " " + kb.width() + "x" + kb.height() + "\n";
报告 += "点前有面板: " + 面板开了() + "\n";

// 实验一：空框状态点它，看面板出不出来
if (面板开了()) {
    键.click();
    sleep(1200);
    报告 += "先收掉已开面板: " + (!面板开了()) + "\n";
}

键.click();
sleep(1300);
var 开了 = 面板开了();
报告 += "实验：点右侧节点 → 面板出现 " + 开了 + "\n";

if (开了) {
    // 点中加号了，同位置再点一下收回（此时图标变 ×）
    var 关 = 找右侧节点();
    if (关) { 关.click(); sleep(1300); }
    报告 += "收回后还有面板: " + 面板开了() + "\n";
} else {
    报告 += "没出现面板，可能是纸飞机状态或节点不可点\n";
}

// 填话术
输入.click();
sleep(900);
try { 输入.setText(话术); } catch (e) { 报告 += "setText 异常: " + e + "\n"; }
sleep(2200);
报告 += "话术已填入，准备点发送位\n";

// 实验二：填完再找同位置节点点下去
var 已发 = false;
for (var r = 1; r <= 3 && !已发; r++) {
    var 新 = 找右侧节点();
    if (!新) { 报告 += "第" + r + "次没扫到右侧节点\n"; break; }
    var nb = 新.bounds();
    报告 += "第" + r + "次节点 @" + (nb.left + nb.width() / 2) + "," + (nb.top + nb.height() / 2);
    新.click();
    sleep(2100);
    已发 = 发出去了();
    报告 += " → 发出: " + 已发 + "\n";
    if (!已发 && 面板开了()) {
        var 收 = 找右侧节点();
        if (收) { 收.click(); sleep(1200); }
        报告 += "  面板又开了，已收起\n";
    }
}

sleep(1000);
setClip(报告);
toastLog(已发 ? "发出去了" : "没发出去");
这版是两个实验，报告能直接看出节点点不点得动
