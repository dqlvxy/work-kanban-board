上一版是从聊天页起跑的，右上角那个「...」点开成了不感兴趣选项。这版先用返回键退回首页，认准底部「职位」页签，再点「我的」。

点下面按钮，脚本进剪贴板

2

AutoJs6 新建 17.js，粘贴，保存

3

直接运行，不用管当前停在哪页

4

把剪贴板报告粘给我

脚本全文（按钮失灵时手动复制）
/**
 * 手机自动投递 · 第 17 步：先退首页，再走 我的 → 设置 → 打招呼语
 */

var 话术 = "杜全龙，18年ICT/FCT治具和FPC连接器治具机械设计，熟SolidWorks，做过手机FCT测试设备、保压盒机构、机器人上下料，服务过华为、OPPO、比亚迪、小米。贵司岗位与我履历很匹配，方便发简历给您吗？";

auto.waitFor();
sleep(800);

var 报告 = "";
launchApp("BOSS直聘");
sleep(4500);
报告 += "包名: " + currentPackage() + "\n";

function 底部有(t) {
    var 全 = text(t).find();
    for (var i = 0; i < 全.size(); i++) {
        if (全.get(i).bounds().top > device.height * 0.88) return 全.get(i);
    }
    return null;
}

function 看文字() {
    var 全 = className("android.widget.TextView").find();
    var 行 = [];
    for (var i = 0; i < 全.size(); i++) {
        var t = (全.get(i).text() || "").trim();
        if (t && t.length < 30) 行.push(t);
    }
    return 行.join(" / ");
}

// 1) 退回首页：最多按 5 次返回
var 试 = 0;
while (试 < 5 && !底部有("职位")) {
    back();
    sleep(1400);
    试++;
}
报告 += "退回次数: " + 试 + "  首页tab: " + (底部有("职位") ? "有" : "没有") + "\n";

// 2) 点「我的」
var 我的 = 底部有("我的");
if (我的) { 我的.click(); sleep(2200); 报告 += "已进我的页\n"; }
else { 报告 += "点不到我的\n"; }

报告 += "我的页文字: " + 看文字() + "\n";

// 3) 我的页右上角设置
var 齿 = null;
var 全N = classNameMatches(/.+/).find();
for (var j = 0; j < 全N.size(); j++) {
    var x = 全N.get(j);
    var b = x.bounds();
    if (!x.clickable()) continue;
    if (b.top > device.height * 0.18) continue;
    if (b.centerX() < device.width * 0.78) continue;
    if (b.width() < 40) continue;
    齿 = x;
}
if (齿) {
    var 齿B = 齿.bounds();
    齿.click(); sleep(2400);
    报告 += "点设置 @ " + 齿B.centerX() + "," + 齿B.centerY() + "\n";
} else 报告 += "没找到设置按钮\n";

报告 += "设置页文字: " + 看文字() + "\n";

// 4) 找打招呼语
var 招 = textContains("打招呼").findOne(3000);
if (招) {
    招.click(); sleep(2000);
    报告 += "已进打招呼语页\n";
    报告 += "打招呼页文字: " + 看文字() + "\n";

    var 开关 = className("android.widget.Switch").findOne(2500);
    if (开关) {
        var 态 = (开关.text() || 开关.desc() || "未知");
        报告 += "开关: " + 态 + "\n";
        if (态 != "已开启" && 态 != "开") { try { 开关.click(); sleep(1200); } catch (e) {} }
    }

    if (!textContains("保存").exists()) {
        var 编 = text("编辑").findOne(2000) || textContains("自定义").findOne(2000) ||
                 text("修改").findOne(2000) || text("设置打招呼语").findOne(2000);
        if (编) { 编.click(); sleep(1800); 报告 += "进编辑\n"; }
    }

    var 框 = className("android.widget.EditText").findOne(4000);
    if (框) {
        try { 框.setText(话术); } catch (e) { 报告 += "setText 异常: " + e + "\n"; }
        sleep(1600);
        报告 += "写入 " + 话术.length + " 字\n";
        报告 += "写入后文字: " + 看文字() + "\n";
        if (text("保存").findOne(2500)) {
            text("保存").findOne(2500).click(); sleep(1500);
            报告 += "已点保存\n";
        } else if (text("确定").findOne(2000)) {
            text("确定").findOne(2000).click(); sleep(1500);
            报告 += "已点确定\n";
        } else {
            报告 += "没找到保存按钮\n";
        }
    } else {
        报告 += "没找到编辑框\n";
    }
} else {
    报告 += "设置页没有打招呼语入口\n";
}

sleep(1000);
setClip(报告);
toastLog("完事");
每走一步都把当页文字记进报告
