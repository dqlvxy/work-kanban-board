脚本全文（按钮失灵时手动复制）
/**
 * 手机自动投递 · 第 5 步：扫岗位详情页
 * 用法：先在 BOSS 里点进一个岗位、停在能看到「立即沟通」的界面，
 *      然后按 Home 退出到桌面，再来运行本脚本。脚本会把 BOSS 拉回那一页并扫描。
 */
auto.waitFor();
sleep(800);

var ok = launchApp("BOSS直聘");
toastLog("拉起 BOSS: " + ok);
sleep(6000);

var 包 = currentPackage();
toastLog("当前前台: " + 包);

var 全部 = classNameMatches(/.+/).find();
var 行 = [];
for (var i = 0; i < 全部.size(); i++) {
    var w = 全部.get(i);
    var 文 = (w.text() || "").trim();
    var 述 = (w.desc() || "").trim();
    if (!文 && !述) continue;
    var b = w.bounds();
    行.push(w.className().replace("android.widget.", "").replace("android.view.", "") +
        " | " + 文 + " | " + 述 + " | " +
        b.left + "," + b.top + " " + b.width() + "x" + b.height());
}

var 内容 = "包名: " + 包 + "\n节点总数: " + 全部.size() +
    "\n含文字: " + 行.length + "\n----\n" + 行.join("\n");

setClip(内容);
toastLog("已复制到剪贴板 " + 行.length + " 条");
顺序很重要：先把 BOSS 停在详情页，再跑脚本
