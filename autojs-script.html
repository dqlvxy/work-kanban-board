/**
 * 手机自动投递 · 第 10 步：网格扫发送键
 * 先把输入框点一下让键盘弹出，再以输入框为基准在右侧网格里逐点试探。
 */

auto.waitFor();
sleep(800);

var ok = launchApp("BOSS直聘");
sleep(4500);

var 包 = currentPackage();
var 报告 = "包名: " + 包 + "  屏宽: " + device.width + "  屏高: " + device.height + "\n";

if (包 != "com.hpbr.bosszhipin") {
    setClip(报告 + "不在 BOSS 前台");
    toastLog("不在 BOSS");
} else {
    var 输入 = className("android.widget.EditText").findOne(6000);
    if (!输入) {
        setClip(报告 + "没找到输入框");
        toastLog("没找到输入框");
    } else {
        输入.click();
        sleep(2200);

        var 原有 = (输入.text() || "").trim();
        var 框 = 输入.bounds();
        报告 += "字数: " + 原有.length + "\n输入框: " + 框.left + "," + 框.top +
            " " + 框.width() + "x" + 框.height() + "\n";

        if (原有.length < 10) {
            setClip(报告 + "输入框是空的，先跑第 7 步填字");
            toastLog("输入框是空的");
        } else {
            var 已发 = false;

            function 清空了吗() {
                var t = (输入.text() || "").trim();
                return t.length < 8;
            }

            // 第一轮：右侧可点节点
            var 全部 = classNameMatches(/.+/).find();
            var 候选 = [];
            for (var i = 0; i < 全部.size(); i++) {
                var w = 全部.get(i);
                var b = w.bounds();
                if (b.width() <= 0 || b.height() <= 0) continue;
                if (b.centerX() < device.width * 0.82) continue;
                if (Math.abs(b.centerY() - 框.centerY()) > 200) continue;
                候选.push(w);
            }
            报告 += "右侧节点 " + 候选.length + " 个\n";

            for (var k = 0; k < 候选.length && !已发; k++) {
                var bb = 候选[k].bounds();
                候选[k].click();
                sleep(1400);
                if (清空了吗()) {
                    已发 = true;
                    报告 += "节点点击命中: " + bb.centerX() + "," + bb.centerY() + "\n";
                }
            }

            // 第二轮：网格试探
            if (!已发) {
                var xs = [device.width - 120, device.width - 100, device.width - 80,
                          device.width - 140, device.width - 60];
                var ys = [框.centerY() - 80, 框.centerY() - 40, 框.centerY(),
                          框.centerY() + 40, 框.centerY() + 80, 框.centerY() + 120];
                for (var a = 0; a < ys.length && !已发; a++) {
                    for (var c = 0; c < xs.length && !已发; c++) {
                        click(xs[c], ys[a]);
                        sleep(1100);
                        if (清空了吗()) {
                            已发 = true;
                            报告 += "网格命中: " + xs[c] + "," + ys[a] + "\n";
                        }
                    }
                }
            }

            if (!已发) 报告 += "两轮都没命中\n";
            sleep(1200);
            setClip(报告);
            toastLog(已发 ? "发出去了" : "还是没发出去");
        }
    }
}
