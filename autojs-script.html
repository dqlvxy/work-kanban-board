脚本全文（按钮失灵时手动复制）
/**
 * 手机自动投递 · 第 11 步：清空 → 粘贴 → 发送
 * 先清空输入框，再用剪贴板粘贴（让 App 感知到真实输入），最后用触摸手势点发送。
 */

var 话术 = "孙先生您好，我是杜全龙，18 年 ICT/FCT 测试治具与 FPC 连接器治具机械设计经验，熟练使用 SolidWorks，做过手机 FCT 测试设备、保压盒缓存机构、双车床机器人上下料等项目，服务过华为、OPPO、比亚迪、小米等客户。看到贵司招机械/结构工程师，5-10 年经验、大专，跟我的履历对得上。方便的话我把简历发您看下？";

auto.waitFor();
sleep(800);

var 报告 = "";
var ok = launchApp("BOSS直聘");
sleep(4500);

if (currentPackage() != "com.hpbr.bosszhipin") {
    setClip("不在 BOSS 前台");
    toastLog("不在 BOSS");
    exit();
}

var 输入 = className("android.widget.EditText").findOne(6000);
if (!输入) {
    setClip("没找到输入框");
    toastLog("没找到输入框");
    exit();
}

// 1) 清空
输入.click();
sleep(600);
try { 输入.setText(""); } catch (e) {}
sleep(500);
try { 输入.setText(""); } catch (e) {}
sleep(800);
报告 += "清空后字数: " + ((输入.text() || "").trim()).length + "\n";

// 2) 粘贴
setClip(话术);
sleep(500);
var 填好 = false;
try {
    var 框 = 输入.bounds();
    longClick(框.centerX(), 框.centerY());
    sleep(1500);
    var 粘 = textContains("粘贴").findOne(2000);
    if (粘) {
        粘.click();
        sleep(1200);
    }
} catch (e) {
    报告 += "长按粘贴报错: " + e + "\n";
}

var 现字 = (输入.text() || "").trim();
if (现字.length >= 100) {
    填好 = true;
    报告 += "剪贴板粘贴成功，字数 " + 现字.length + "\n";
} else {
    try { 输入.setText(话术); } catch (e) {}
    sleep(1200);
    现字 = (输入.text() || "").trim();
    报告 += "退回 setText，字数 " + 现字.length + "\n";
    if (现字.length >= 100) 填好 = true;
}

if (!填好) {
    setClip(报告 + "话术没填进去，停止");
    toastLog("话术没填进去");
    exit();
}

sleep(1200);
var 框2 = 输入.bounds();

function 清空了吗() {
    return ((输入.text() || "").trim()).length < 8;
}

function 触摸点(x, y) {
    try { swipe(x - 4, y - 4, x + 4, y + 4, 70); } catch (e) {}
}

var 已发 = false;

// 3a) 右侧节点直点
var 全部 = classNameMatches(/.+/).find();
var 候选 = [];
for (var i = 0; i < 全部.size(); i++) {
    var w = 全部.get(i);
    var b = w.bounds();
    if (b.width() <= 0 || b.height() <= 0) continue;
    if (b.centerX() < device.width * 0.80) continue;
    if (Math.abs(b.centerY() - 框2.centerY()) > 260) continue;
    候选.push(w);
}
报告 += "右侧节点 " + 候选.length + " 个\n";
for (var k = 0; k < 候选.length && !已发; k++) {
    var bb = 候选[k].bounds();
    候选[k].click();
    sleep(1300);
    if (清空了吗()) {
        已发 = true;
        报告 += "节点命中 " + bb.centerX() + "," + bb.centerY() + "\n";
    }
}

// 3b) 网格 触摸点（不是 click，走手势）
if (!已发) {
    var xs = [device.width - 130, device.width - 110, device.width - 95,
              device.width - 75, device.width - 55];
    var ys = [框2.centerY() - 90, 框2.centerY() - 45, 框2.centerY(),
              框2.centerY() + 45, 框2.centerY() + 90, 框2.centerY() + 135];
    for (var a = 0; a < ys.length && !已发; a++) {
        for (var c = 0; c < xs.length && !已发; c++) {
            触摸点(xs[c], ys[a]);
            sleep(1100);
            if (清空了吗()) {
                已发 = true;
                报告 += "触摸命中 " + xs[c] + "," + ys[a] + "\n";
            }
        }
    }
}

// 3c) 还不行就用 click 再补一遍中心线
if (!已发) {
    for (var m = -120; m <= 160 && !已发; m += 40) {
        click(device.width - 100, 框2.centerY() + m);
        sleep(1000);
        if (清空了吗()) {
            已发 = true;
            报告 += "click 命中 " + (框2.centerY() + m) + "\n";
        }
    }
}

报告 += "输入框: " + 框2.left + "," + 框2.top + " " + 框2.width() + "x" + 框2.height() + "\n";
if (!已发) 报告 += "三轮都没命中\n";

sleep(1000);
setClip(报告);
toastLog(已发 ? "发出去了" : "还是没发出去");
跑完框里空了就是发出去了
