/**
 * 手机自动投递 · 第 31 步：先把 BOSS 拉到前台，再倒库 + 试点会话行
 * v30 教训：脚本没 launchApp，AutoJs6 自己在前台，扫到的 21 条全是控制台日志。
 */

var PKG = "com.hpbr.bosszhipin";

auto.waitFor();
sleep(600);
var 报告 = "";

function 宽(b) { return b.right - b.left; }
function 高(b) { return b.bottom - b.top; }
function 心X(b) { return (b.left + b.right) / 2; }
function 心Y(b) { return (b.top + b.bottom) / 2; }

// 把 BOSS 拉到前台，拉两次确保成功
launchApp("BOSS直聘");
sleep(4500);
if (currentPackage() != PKG) {
    launchApp("BOSS直聘");
    sleep(4000);
}
报告 += "前台 " + currentPackage() + "  屏 " + device.width + "x" + device.height + "\n";
if (currentPackage() != PKG) {
    setClip(报告 + "\nBOSS 没被拉到前台，脚本停手");
    toastLog("BOSS 没到前台");
    exit();
}
sleep(1200);

// ———— 1. 只读 BOSS 自己的节点 ————
var 项 = [];
packageName(PKG).classNameMatches(/.+/).find().forEach(function (n) {
    if (项.length > 2600) return;
    try {
        var t = n.text() || "", d = n.desc() || "", cl = n.clickable();
        if (!t && !d && !cl) return;
        var b = n.bounds();
        项.push({ y: 心Y(b), s: (cl ? "[可点]" : "") + (n.className() || "?") +
            " 文=" + (t || "-") + " 描=" + (d || "-") +
            " @" + Math.round(心X(b)) + "," + Math.round(心Y(b)) + " " + 宽(b) + "x" + 高(b) });
    } catch (e) { }
});
项.sort(function (a, b) { return a.y - b.y; });
报告 += "\n节点共 " + 项.length + " 条（前 50）\n";
for (var i = 0; i < 项.length && i < 50; i++) 报告 += 项[i].s + "\n";

// ———— 2. 会话行候选 ————
var 行 = [];
packageName(PKG).clickable(true).find().forEach(function (n) {
    var b = n.bounds();
    var w = 宽(b), h = 高(b);
    if (w > 500 && h > 60 && h < 420) 行.push(n);
});
行.sort(function (a, b) { return 心Y(a.bounds()) - 心Y(b.bounds()); });
报告 += "\n会话行候选 " + 行.length + " 个\n";
行.forEach(function (n, k) {
    var b = n.bounds();
    报告 += "  第" + (k + 1) + "行 @" + Math.round(心X(b)) + "," + Math.round(心Y(b)) + " " + 宽(b) + "x" + 高(b) +
        " 文=" + (n.text() || "-") + " 描=" + (n.desc() || "-") + "\n";
});

// ———— 3. 点第 3 行 ————
var 进 = false;
if (行.length >= 3) {
    报告 += "\n点第 3 行…";
    行[2].click();
    sleep(2900);
    var e = className("android.widget.EditText").findOnce();
    进 = e != null && e.bounds().top > device.height * 0.70;
    报告 += " 前台 " + currentPackage() + " 输入框y=" + (e ? e.bounds().top : "无") + " 进聊天页 " + 进 + "\n";
} else if (行.length > 0) {
    报告 += "\n会话行不够 3 个，点第 1 行试试…";
    行[0].click();
    sleep(2900);
    报告 += " 前台 " + currentPackage() + "\n";
} else {
    报告 += "\n没读到会话行\n";
}

sleep(500);
setClip(报告);
toastLog("已复制");
