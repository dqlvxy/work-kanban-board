/**
 * 手机自动投递 · 第 21 步：找到「我的」页签并摸清设置入口
 * 上一版教训：back 次数多了会把 BOSS 退掉，脚本跑到 AutoJs6 界面里乱点。
 * 这版每步都校验前台包名，一旦掉出 BOSS 立刻 launcher 拉回来。
 */

var PKG = "com.hpbr.bosszhipin";
auto.waitFor();
sleep(800);
var 报告 = "";

function 在BOSS() { return currentPackage() === PKG; }

function 回BOSS() {
    for (var i = 0; i < 4 && !在BOSS(); i++) {
        app.launchPackage(PKG);
        sleep(2800);
    }
    return 在BOSS();
}

function 抓页(标签) {
    if (!在BOSS()) { 报告 += "\n===== " + 标签 + " =====\n（不在 BOSS 前台，跳过）\n"; return; }
    var 文 = [], 钮 = [];
    packageName(PKG).className("android.widget.TextView").find().forEach(function (n) {
        var t = n.text();
        if (t && t.length) 文.push(t);
    });
    packageName(PKG).find().forEach(function (n) {
        if (!n.clickable()) return;
        var d = (n.desc() || "") + "|" + (n.text() || "");
        var b = n.bounds();
        if (b.width() > 0 && b.height() > 0) 钮.push(d + "@" + (b.left + b.width() / 2) + "," + (b.top + b.height() / 2));
    });
    报告 += "\n===== " + 标签 + " =====\n";
    报告 += "文字 " + 文.length + ": " + 文.join(" / ") + "\n";
    报告 += "可点 " + 钮.length + ": " + 钮.slice(0, 45).join("  ") + "\n";
}

回BOSS();
sleep(1500);
报告 += "启动后前台: " + currentPackage() + "\n";

// 找底部「我的」页签：最多 back 4 次，每次校验包名
var 找到我的 = false;
for (var i = 0; i < 4 && !找到我的; i++) {
    if (!回BOSS()) { 报告 += "第" + (i+1) + "轮拉不回 BOSS\n"; break; }
    var 我 = text("我的").findOnce(600);
    if (我 && 我.bounds().top > device.height * 0.82) { 找到我的 = true; 报告 += "底部见到「我的」页签\n"; break; }
    报告 += "第" + (i+1) + "轮没见底部页签，按返回\n";
    back();
    sleep(1600);
    报告 += "  返回后前台: " + currentPackage() + "\n";
}

if (!找到我的) {
    抓页("当前位置");
    setClip(报告 + "\n没摸到底部「我的」页签");
    toastLog("没摸到我的页签");
    exit();
}

var 我 = text("我的").findOnce(600);
var p = 我;
for (var k = 0; k < 4 && p && !p.clickable(); k++) p = p.parent();
if (!p) p = 我;
p.click();
sleep(2500);
报告 += "点完「我的」前台: " + currentPackage() + "\n";
抓页("我的页");

// 我的页里找设置类入口
var 候选 = ["设置", "打招呼", "隐私"];
for (var c = 0; c < 候选.length; c++) {
    var n = textContains(候选[c]).findOnce(400);
    if (n) {
        var q = n;
        for (var j = 0; j < 4 && q && !q.clickable(); j++) q = q.parent();
        if (q) {
            q.click();
            sleep(2200);
            报告 += "点了「" + 候选[c] + "」\n";
            抓页("点了" + 候选[c] + "之后");
            back();
            sleep(1500);
            回BOSS();
            break;
        }
    }
}

setClip(报告);
toastLog("结构已复制到剪贴板");
